\textit{Sea $f:\R^2\rightarrow \R^2$ tal que para cada $x\in \R^2$:
\begin{equation*}
    f(x)=e^{x_1}\,\begin{pmatrix}
        \cos(x_2)\\
        \sin(x_2)
    \end{pmatrix}
\end{equation*}
Determinar si $f(\R\times [0,2\pi))$ es un conjunto abierto, cerrado, acotado, compacto.}
\end{tcolorbox}
\begin{proof}\,\\
  Sea $x\in \R^2\setminus \{0_{\R_2}\}$, sabemos que podemos representarlo de la forma $x=r\,\begin{pmatrix}
        \cos(\theta)\\
        \sin(\theta)
  \end{pmatrix}$, donde $r\in \R^+$, y $\theta\in [0,2\pi)$, sabemos que existe $x_1\in \R$ tal que $e^{x_1}=r$, por tanto $f(x_1,\theta_1)=x$, por tanto $\R^2\setminus \{0_{\R^2}\}\subset f(\R\times [0,2\pi))$, Luego
  basta notar que como $e^y>0$ para toda $y\in \R$, entonces $f(m)\neq 0_{\R^2}$ para todo $m\in\R^2$, por tanto $f(\R\times[0,2\pi))=\R^2\setminus\{0_{R_2}\}$, el cual claramente es abierto, pues su complemento es el conjunto $\{0_{R^2}\}$ el cual claramente es cerrado, como es abierto no puede ser cerrado ni compacto, y finalmente no es
  acotado pues $\R^2$ no es acotado.
\end{proof}

\textit{
Sea $\chi$ un conjunto no vacío. Decimos que una colección $\zeta\subset 2^{\chi}$ cumple la propiedad de la intersección finita cuando para cada $A\subset \zeta$ de cardinalidad finita se cumple que:
\begin{equation*}
    \bigcap_{B\in A}\,B \neq \emptyset
\end{equation*}
Demostrar que si $A\subset \mathbb{R}^n$ es compacto, entonces para cada $\chi\subset 2^A$ que cumpla la propiedad de la intersección finita y que para cada $B\in \chi$ existe $U_{B}\in \mathbb{R}^n$ conjunto cerrado tal que $B=U_{B}\cap A$, se cumple que:
\begin{equation*}
    \bigcap_{C\in\chi}\,C\neq \emptyset
\end{equation*}
}
\end{tcolorbox}
\begin{proof}\,\\
    Procedemos por contradicci\'on, es decir existe $\chi\subset2^A$ tal que cumple la propiedad de la intersecci\'on finita y que para todo $B\in \chi$ existe $U_B\in \R^n$ cerrado tal que $B=U_B\cap A$, y ademas:\,\\
    \begin{equation*}
      \bigcap_{B\in \chi}\,B=\emptyset
    \end{equation*}\,\\
    Si sacamos el complemento respecto a $A$\,\\
    \begin{equation*}
      A=(\emptyset)^c=\left(\bigcap_{B\in \chi}\,B\right)^c=\bigcup_{B\in \chi}\,(U_B\cap A)^c
    \end{equation*}\,\\
    Como $U_B$ es cerrado para todo $B\in \chi$ y $A$ es compacto en $\R^n$ se tiene que $U_B\cap A$ es cerrado por tanto $(U_B\cap A)^c$ es abierto, por tanto $\displaystyle \bigcup_{B\in \chi}\,(U_B\cup A)^c$ es una cubierta abierta de $A$, usando la compacidad de $A$ decimos que existe
    $\zeta\subset \chi$ de cardinalidad finita tal que:\,\\
    \begin{equation*}
      A=\bigcup_{B\in \zeta}\,(U_B\cap A)^c
    \end{equation*}\,\\
    De nuevo sacando el complemento respecto a $A$:\,\\
    \begin{equation*}
      \emptyset=\left(\bigcup_{B\in \zeta}\,(U_B\cap A)^c\right)^c=\bigcap_{B\in \zeta}\,(U_B\cap A)=\bigcap_{B\in \zeta}\,B
    \end{equation*}\,\\
    Pero como $\chi$ cumple la propiedad de la intersecci\'on finita y $\zeta\subset\chi$ se sigue que:\,\\
    \begin{equation*}
      \emptyset=\bigcap_{B\in \zeta}\,B\neq \emptyset
    \end{equation*}\,\\
    Una contradicci\'on , como nuestra unica suposici\'on fue la negaci\'on de la tesis del problema se concluye la veracidad de la misma
   

\end{proof}

\textit{Demostrar por definición que ser sucesión de Cauchy es equivalente a ser una sucesión convergente en $\R^n$.}
\end{tcolorbox}
\begin{proof}\,\\
  $\Rightarrow )$\,Supongamos que $\{a_n\}_{n\in \N}\subset \R^m$ es convergente a $l\in \R^m$, entonces para $\frac{\epsilon}{2}>0$, existe $k\in \N$ tal que si $m,n>k$ entonces
  $\|a_n-l\|<\frac{\epsilon}{2}$, $\|l-a_m\|\leq \frac{\epsilon}{2}$, por tanto:\,\\
  \begin{equation*}
   =\|a_n-a_m\|= \|a_n-l+l-a_m\|\leq \|a_n-l\|+\|l-a_m\|<\frac{\epsilon}{2}+\frac{\epsilon}{2}=\epsilon
  \end{equation*}\,\\
  Por tanto $\{a_n\}_{n\in \N}\subset \R^n$ es sucesi\'on de Cauchy.\,\\
  \,\\
  $\Longleftarrow )$Si $\{a_n\}_{n\in \N}\subset \R^m$ esto implica que todas sus componetes $\{(a_{i})_n\}_{n in\N }\subset \R$, con $i\in \{1,2,\cdots,m\}$ son sucesiones de cauchy y por tanto convergentes en $\R$, como cada componente converge, tenemos que la sucesi\'on original $\{a_n\}_{n\in N}\subset R^m$ converge

\end{proof}

\textit{Sea $f:A\subset \R^n\rightarrow \R^m$ una función. Demostrar que $f$ es continua si y solo si para cada $C\in\R^{m}$, cerrado en $\R^m$, existe $B\subset \R^n$ tal que $f^{-1}(C)=B\cap A$.}
\end{tcolorbox}
\begin{proof}\,\\
  $\Rightarrow)$\, Supongamos que $f$ es continua, sea $B\subset \R^m$ cerrado, sabemos que $B^c$ es abierto en $\R^m$ y por la fontinuidad de $f$ existe $C\in \R^n$ abierto tal que:
  \begin{equation*}
    f^{-1}(B^c)=A\cap C\implies (f^{-1}(B))^c\cap A=C\cap A\implies f^{-1}(B)\cup A^c=C^c\cup A^c
  \end{equation*}\,\\
  Luego se sigue que:\,\\
  \begin{equation*}
    (f^{-1}(B)\cup A^c)\cap A=(C^c\cup A^c)\cap A\implies f^{-1}(B)=f^{-1}(B)\cap A=C^c\cap A
  \end{equation*}\,\\
  Con $C^c$ cerrado en $\R^n$\,\\
  \,\\
  $\Leftarrow)$\, Ahora supongamos que sea $B\subset \R^m$ cerrado, entonces existe $C\in \R^n$ tal que $f^{-1}(B)=C\cap A$, tomemos $D\subset R^m$ abierto, entonces $D^c$ es cerrado en $\R^m$ por tanto, existe $E\in \R^n$ cerrado tal que:\,\\
  \begin{equation*}
    f^{-1}(D^c)=E\cap A\implies (f^{-1}(D))^c\cap A=E\cap A\implies f^{-1}(D)\cup A^c=E^c\cup A^c 
  \end{equation*}\,\\
  Luego se sigue que:\,\\
  \begin{equation*}
    (f^{-1}(D)\cup A^c)\cap A=(E^c\cup A^c)\cap A\implies f^{-1}(D)=f^{-1}(D)\cap A=E^c\cap A
  \end{equation*}\,\\
  Con $E^c$ abierto relativo en $A$, como $D$ fue un abierto de $\R^m$, tenemos que $f$ es continua.
\end{proof}
