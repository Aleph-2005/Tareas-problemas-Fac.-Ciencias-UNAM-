\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 1}},]
\textit{Sea $G$ un grupo de orden $p^3$, con $p$ primo. Demuestra que si $G$ no es abeliano, entonces
\begin{equation*}
    |Z(G)|=p
\end{equation*}
} 
\end{tcolorbox}
\begin{proof}
    Tenemos que como $G$ es $p$-grupo entonces $Z(G)$ es no trivial, como este es un subgrupo por teorema de lagrange tenemos que $|Z(G)|\in\{p,p^2\}$, si $|Z(G)|=p^2$ entonces
    $[G:Z(G)]=\frac{p^2}{p}=p$, por tanto $G\setminus Z(G)$ es ciclico, pues es de orden primo, ahora por un problema anterior sabemos que si el cociente es ciclico entonces el grupo es abeliano, por tanto $G$ seria abeliano contradiciendo la hip\'otesis, es decir solo es posible que
    $|Z(G)|=p$
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}},]
\textit{Demuestra que todo grupo de orden $6$ es abeliano o isomorfo a $S_6$
} 
\end{tcolorbox}
\begin{proof}
    Supongamos que $G$ es no abeliano, por tanto no existe ningun elemento de orden $6$. tomemos $a\in G$ de orden $3$, este genera un subgrupo de indice $2$, que es normal, no puede haber dos subgrupos diferentes $H$ y $K$ de orden tres pues $HK$ seria de orden $9$, asi que tomemmos $H\subseteq G$ de orden $3$ sea $x\in G\setminus H$, entonces $\langle x \rangle$ tiene ordenn dos necesariamente 
    por teorema de Caley generalizado existe $\Delta:G\to S_{G\setminus H}$, tal que $Ker(\Delta)$ es el mayor subgrupo normal de $G$ contenido en $\langle x \rangle$, como $\langle x \rangle$ es de orden $2$ y no es normal, entonces el nucleo de $\Delta$ es $\{e\}$, ie $\Delta$ es monomorfismo entre $G$ y $S_3$, por tanto $G\equiv S_3$, pues $G$ y $S_3$ son ambos finitos, entonces cualquier monomorfismo entre ellos es isomorfismo.
\end{proof}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 3}},]
\textit{Sea $G$ un grupo finito, tal que:\,\\
\begin{equation*}
    |G|=p^im,\,\,\text{con}\,\,m>1
\end{equation*}\,\\
donde $p$ es primo $1\leq i$ y $(p,m)=1$
Demuestra que si $H\leq G$ tiene orden $|H|=p^k$ con $0\leq k<i$, entonces existe un subgrupo $K\leq G$ tal que:\,\\
\begin{equation*}
    H\unlhd K\,\,\text{y}\,\,H\subset K
\end{equation*}
} 
\end{tcolorbox}
\begin{proof}
    Tomemos $H<G$ con $|H|=p^k$ con $0\leq k<i$, entonces $[G:H]=p^{i-k}m$ tenemos que $i-k>0$ por tanto $p|[G:H]$, ahora tenemos que:
    \begin{equation*}
        [G:H]\equiv [N_G(H):H] mod\,\,p \implies p\equiv [N_G(H):H] mod\,\,p
    \end{equation*}
    Donde $N_{G}(H)$ es el normalizador de $H$ en $G$, como $p|[[N_G(H):H]]$ esto implica que $[N_G(H):H]>1$ ie $H\neq N_G(H)$
    asi que $N_G(H)\leq G$, $H\subset N_G(H)$ y $H\unlhd N_G(H)$
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 4}},]
\textit{Sea $D_4=\{r,s:r^4=e,s^2=e,srs=r^{-1}\}$ el grupo di\'edrico de orden $8$. Considera el conjunto:
\begin{equation*}
    X=\{1,2,3,4\}
\end{equation*}\,\\
Define una acci\'on dde $D_4$ en $X$ mediante:\,\\
\begin{equation*}
    r\rightarrow^{f}(1,2,3,4),\,\,\,s\rightarrow^{f}(2,4)
\end{equation*}\,\\
\begin{itemize}
    \item Verifica que esta definici\'on es una acci\'on de grupo. Es decir, que $f$ se puede extender a un morfismo $D_4\rightarrow S_X$
    \item Describe las \'orbitas de los elementos $X$
    \item Calcula los estabilizadores de cada elemento
    \item Verifica que se cumple la f\'ormula \'orbita-estabilizador:\,\\
    \begin{equation*}
        |D_4|=|Orb(X)||Stab(X)|
    \end{equation*}
    \item Determina si la acci\'on es fiel y si es transitiva
\end{itemize}
} 
\end{tcolorbox}
\begin{Solucion}
    Tenemos que \(D_4=\{e,r,r^2,r^3,s,rs,sr,r^2s\}\). Pero también
    \[(1234)^4=(1), (24)^2=(1), (24)(1234)(24)=(4321)=(1234)^{-1}.\]
    Por lo tanto, \(D_4 \cong \langle (1234), (24) \rangle\). Ahora si \(f : D_4 \rightarrow \langle (1234), (24) \rangle\) es el isomorfismo dado por 
    \[f(r)=(1234), f(s)=(24),\] entonces se tiene que:
    \begin{align*}
        &f(e)=f(r^4)=(1234)^4=(1), \\
        &f(r^2)=(1234)^2=(13)(24), \\
        &f(r^3)=(1234)^3=(4321), \\
        &f(s)=(24), \\
        &f(sr)=(1234)(24)=(14)(23), \\
        &f(rs)=(24)(1234)=(12)(34), \\
        &f(r^2s)=(13)(24)(1234)=(13).
    \end{align*}
    Por otra parte \[ \mathrm{Orb}(1)=\{ f(g)(1) \mid g \in D_4\} =\{ f(e)(1),f(r)(1), f(r^2)(1), f(r^3)(1), \ldots \}=\{1,2,3,4\}= X. \]
    Como las orbitas forman una partición de \(X\) y \(X\) es la órbita de \(1\), entonces 
    \(X=\mathrm{Orb}(1)=\mathrm{Orb}(2)=\mathrm{Orb}(3)=\mathrm{Orb}(4)\) es la única órbita. Por lo tanto, la acción es transitiva. Además,
    \begin{align*}
        &\mathrm{Stab}(1)=\{g \in D_4 \mid f(g)(1)=1\}=\{e,s \}, \\
        &\mathrm{Stab}(2)=\{g \in D_4 \mid f(g)(2)=2\}=\{e,r^2s\}, \\
        &\mathrm{Stab}(3)=\{g \in D_4 \mid f(g)(3)=3\}=\{e,s\}, \\
        &\mathrm{Stab}(4)=\{g \in D_4 \mid f(g)(4)=4\}=\{e,r^2s\}.
    \end{align*}
    Por lo tanto, \(|\mathrm{Stab}(i)|=2\) para cada \(i\in \{1,2,3,4\}\). Se cumple el teorema de órbita-estabilizador
    \[|D_4|=|\mathrm{Orb}(i)|\cdot |\mathrm{Stab}(i)|=4\cdot 2=8.\]
    Finalmente, como \(\mathrm{Nuc}(f)=\mathrm{Stab}(1)\cap\mathrm{Stab}(2)\cap\mathrm{Stab}(3)\cap\mathrm{Stab}(4)=\{e\}\), se tiene que \(f\) es inyectiva. Y por tanto el fiel.
\end{Solucion}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 5}},]
\textit{Sea $n\geq 5$. Demuestra que $S_n$ no tiene subgrupos de \'indice $t$
con $2<t<n$
} 
\end{tcolorbox}
\begin{proof}
     \begin{lema}
        $A_n$ es subgrupo propio maximal de $S_n$ para toda $n\in \N$ (i.e si $A_n\leq H<S_n$ entonces $A_n=H$)
    \end{lema}
    \begin{proof}
        Supongamos que existe $A_n\leq H<S_n$, tenemos que:\,\\
        \begin{equation*}
            2=[S_n:A_n]=[S_n:H][H:A_n]
        \end{equation*}
        Como $2$ es primo entonces necesariamente $[S_n:H]=2$ y $[H:A_n]=1$ \'o $[S_n:H]=1$ y $[H:A_n]=2$, como $H$ es subgrupo propio entonces necesariamente $[S_n:H]>1$ ie $[H:A_n]=1$ por tanto $H=A_n$
    \end{proof}
    \begin{lema}
        Para $n\geq 5$ los unicos subgrupos normales de $S_n$ son $\{e\},A_n, S_n$
    \end{lema}
    \begin{proof}
        No puede haber un subgrupo normal $H\neq \{e\}$ tal que $H<A_n<S_n$ ya que $H$ seria normal en $A_n$ y por su simplicicdad necesariamente $A_n=H$ o $H=\{e\}$, luego por el lema anterior
        $A_n$ es subgrupo propio maximal, por tanto no puede haber ning\'un subgrupo normal $F$ con $A_n<F<S_n$ concluyendo la prueba 
    \end{proof}
    Supongamos que existe $H<G$ tal que $[G:H]=t$ con $2<t<n$, por teorema de Caley genralizado existe:
    \shorthandoff{<>}
\[
\begin{tikzpicture}[node distance=3cm]
  \node (A) {$G$};
  \node (B) [right of=A] {$\Delta[G]\leq S_{[G:H]=t}$};
  \draw[->] (A) -- (B) node[midway, above] {$\Delta$};
\end{tikzpicture}
\]
Luego $Ker(\Delta)\leq H<S_n$ es normal en $S_n$, luego como los unicos grupos normales de $S_n$ son $\{e\},A_n,S_n$ tenemos que $Ker(\Delta)\in\{A_n,\{e\}\}$, si $Ker(\Delta)=\{e\}$ entonces $\Delta$ es un isomorfismo entre su dominio y su imagen i.e $n!|t\implies n<t<n$ una contradicción, por tanto
$ker(\Delta)=A_n$, como $A_n$ es subgrupo propio maximal de $S_n$ entonces $A_n=H$ ie $t=2>2$ de nuevo una contradicción, como nuestra unica suposici\'on fue la existencia de $H$ concluimos su no existencia 
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 6}},]
\textit{Sea $G$ un grupo finito y sea $p$ el menos primo $p$ que divide a $|G|$. Sup\'on que $H\subseteq G$ satisface:\,\\
\begin{equation*}
    [G:H]=p
\end{equation*}\,\\
Demuestra que $H\unlhd G$
} 
\end{tcolorbox}
\begin{proof}
    Es claro que sea $N_{G}(H)$ el normalizador de $H$ en $G$:\,\\
    \begin{equation*}
        p=[G:H]=[G:N_G(H)][N_G(H):H]
    \end{equation*}
    entonces como $p$ es primo $[G:N_G(H)]=p$ y $[N_G(H):H]=1$ \'o $[G:N_G(H)]=1$ y $[N_G(H):H]=p$, notemos que como:\,\\
    \begin{equation*}
        [G:H]\equiv [N_G(H):H] mod p \implies p\equiv [N_G(H):H] mod p
    \end{equation*}
    ie $p|[N_G(H):H]$ por tanto $1<[N_G(H):H]$ finalmente se sigue que $[G:N_G(H)]=1$ y $[N_G(H):H]=p$ , por tanto $G=N_G(H)$ ie $H\unlhd G$
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 7}},]
\textit{Sea $G$ un grupo de orden $pm$, donde $p$ es primo y $1<m<p$. Demuestre que $G$ no es simple
} 
\end{tcolorbox}
\begin{proof}
    Supongamos que $G$ es simple, tenemos que por teorema de Cauchy existe $H<G$ tal que $|H|=p$, por tanto $[G:H]=m$, por teorema de Caley extendido tenemos que:\,\\
    existe $\Delta$ morfismo de grupos tal que:
    \shorthandoff{<>}
\[
\begin{tikzpicture}[node distance=3cm]
  \node (A) {$G$};
  \node (B) [right of=A] {$\Delta[G]\leq S_{[G:H]=m}$};
  \draw[->] (A) -- (B) node[midway, above] {$\Delta$};
\end{tikzpicture}
\]
Ademas que $Ker(\Delta)\leq H<G$, como $ker(\Delta)\unlhd G$ y $G$ es simple por la condici\'on anterior necesariamente $ker(\Delta)=\{e\}$, pues $H$ es subgrupo propio de $G$, por tanto $\Delta$ es inyectiva y por tanto $G\cong \Delta[G]$, por tanto $pm|m!$, luego $p|(m-1)!$ pues $m>1$, por tanto $p|k$ para alg\'un $k\leq m-1$ ie $p<m$ una contradicción, como nuestra una suposici\'on fue
que $G$ es simple concluimos que $G$ no puede ser simple
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 8}},]
\textit{Demuestra que no hay grupos simples de orden $pq$, con $p$ y $q$ primos.}
\end{tcolorbox}
\begin{proof}
    Si $p=q$ entonces $|G|=p^2$, luego $C(G)\neq\{e\}$, por tanto $|C(G)|=\{p,p^2\}$ si $|C(G)|=p^2$, enconces $|G|=|C(G)|$ ie $G$ es abeliano,  si $|C(G)|=p$ entonces $|G\setminus C(G)|=\frac{p^2}{p}=p$ por tanto $G\setminus C(G)$ es ciclico, ie $G$ es abeliano, como $G$ es abeliano cualquiera de sus subgrupos es normal, por teorema de Cauchy existe $H<G$ con $|H|=p$, ie $G$ tiene un subgrupo normal no trivial, i.e no es simple\,\\
    \,\\
    Ahora si $p\neq q$ sin perdida de generalidad $p<q$ entonces $|G|=qp$ con $1<p<q$ y $q$ primo por el problema anterior $G$ no es simple
\end{proof}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 10}},]
\textit{Sea $x\in G$ 
\begin{itemize}
    \item Demuestra que:
    \begin{equation*}
        C_G(x)\subseteq N_G(\langle x \rangle)
    \end{equation*}
    \item Da un ejemplo donde la inclusi\'on sea estricta
    \item Interpreta ambos subgrupos en t\'erminos de acciones de grupo
\end{itemize}}
\end{tcolorbox}
\begin{proof}
    Tomemos $y\in C_g(x)$, tenemos que:\,\\
    \begin{equation*}
        y^{-1}xy=x\implies \overbrace{y^{-1}xyy^{-1}xy\cdots yy^{-1}xy}^{n\,\,\text{veces}}=y^{-1}x^{n}y=x^{n}\,\,\forall\,n\in \Z
    \end{equation*}\,\\
    es decir para cualquier elemento $x\in \langle x\rangle$ tenemos que $y^{-1}xy=y$ ie $y\in N_G(\langle x \rangle)$, como $y$ fue arbitrario concluimos que $C_{G}(x)\subseteq N_G(\langle x \rangle)$
    \,\\
    \,\\
    Si tomamos $D_4$ el grupo diedrico de las simetrias del cuadrado, tomamos $r$ una rotaci\'on de $90$ grados, sabemos que $srs^{-1}=r^{-1}$ son cualquier reflexi\'on $s$, es claro que $s\notin C_{D_4}(r)$ pues $srs^{-1}=r^{-1}\neq r$ sin embargo $s\in N_{D_4}(r)$ pues
    \begin{equation*}
        s\langle r \rangle s^{-1}=\{e,r^{-1},r^2,r^{-3}\}=\langle r \rangle
    \end{equation*}
    El centralizador $C_G(x)$ es el estabilizador del elemento $x$ bajo la acci\'on de $G$ sobre si mismo bajo conjugaci\'on (i.e $(g,x)\to gxg^{-1}$)\,\\
    \,\\
    El normalizador $N_G(\langle x \rangle)$ es el establizadore de un elemento $x$ cuando el grupo $G$ actua sobre el conjunto de todos sus subgrupos $H$ bajo la accion de conjugaci\'on (i.e $(g,H)\to gHg^{-1}$)
\end{proof}
