\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 1}},]
\textit{Sean $\mathcal{F}_1$ y $\mathcal{F}_2$dos $\sigma$-algebras de $\Omega$. Definimos:\,\\
\,\\
\begin{equation*}
    \mathcal{F}_1\cap \mathcal{F}_2:=\{A\subset \Omega|A\in \mathcal{F}\,\,\,y\,\,\,A\in \mathcal{F} \} 
\end{equation*}
\begin{equation*}
    \mathcal{F}_1 \cup \mathcal{F}_2:=\{A\subset\Omega|A\in \mathcal{F}_1\,\,\,o\,\,\,A\in \mathcal{F}_2\}
\end{equation*}
\begin{itemize}
    \item Demuestre que $\mathcal{F}_1\cap \mathcal{F}_2$ es un $\sigma$-algebra
    \item Mediante un contraejemplo demuestre que $\mathcal{F}_1\cup \mathcal{F}_2$ no necesariamente es una $\sigma$-algebra 
\end{itemize}
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
    i)\,Probaremos que la intersecci\'on de $\sigma$-algebras es una $\sigma$ -algebra probando que este conjunto cumple con las tres propiedades:\,\\
    \begin{enumerate}
        \item  Como $\mathcal{F}_1$ es $\sigma$-algebra entonces $\Omega\in \mathcal{F}_1$, analogamente $\Omega \in \mathcal{F}_2$, por tanto $\Omega\in \mathcal{F}_1 \cap \mathcal{F}_2$
        \item Tenemos que si $A\in \mathcal{F}_1\cap \mathcal{F}_2$, entonces $A\in \mathcal{F}_1$ como $\mathcal{F}_1$ es $\sigma$-algebra por tanto $A^c\in \mathcal{F}_1$, analogamente como $A\in \mathcal{F}_2$ entonces $A^c\in \mathcal{F}_2$ por tanto $A^c\in \mathcal{F}_1\cap \mathcal{F}_2$
        \item Si $\{A_k\}_{k\in \N}\subseteq \mathcal{F}_1\cap \mathcal{F}_2$, entonces en particular $\{A_k\}_{k\in \N}\subseteq \mathcal{F}_1$, entonces como $\mathcal{F}_1$ es $\sigma$-algebra se sigue que $\displaystyle\bigcup_{k\in \N}\,A_k\in \mathcal{F}_1$, analogamente como $\{A_k\}_{k\in \N}\subseteq \mathcal{F}_2$ entonces $\displaystyle \bigcup_{k\in \N}\,A_K\in \mathcal{F}_2$, por tanto
        $\displaystyle \bigcup_{k\in \N}\,A_k\in \mathcal{F}_1\cap \mathcal{F}_2$
    \end{enumerate}\,\\
    Por tanto concluimos que la intersecci\'on de $\sigma$-algebras es $\sigma$-algebra\,\\
    \,\\
    ii)\,Tomemos $\Omega:=\{1,2,3,4,\}$ y $\mathcal{F}_1=\{\{1,2,4\},\{3\},\Omega,\emptyset\}$, $\mathcal{F}_2:=\{\{2,3,4\},\{1\},\Omega,\emptyset\}$, afirmamos que $\mathcal{F}_1$ es $\sigma$-algebra, pues contiene al universo y cualquier elemento contenido en esta cumple que su complemento tambien lo esta, ademas la union entre dos o más elementos siempre da el vacio o el universo, 
    por las mismas razones $\mathcal{F}_2$ es $\sigma$-algebra, luego tenemos que:\,\\
    \begin{equation*}
        \mathcal{F}_1\cup \mathcal{F}_2:=\{\{1,2,4\},\{2,3,4\},\{1\},\{3\},\Omega,\emptyset\}
    \end{equation*}\,\\
    No es $\sigma$-algebra pues $\{1\},\{3\}\in \mathcal{F}_1\cup\mathcal{F}_2$ pero $\{1\}\cup\{3\}=\{1,3\}\notin \mathcal{F}_1\cup \mathcal{F}_2$
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}},]
\textit{Sea $\Omega:=\{1,2,\cdots,n\}$ y considere el espacio medible $(\Omega,2^{\Omega})$. Investigueen cada caso si $\mathbb{P}$
es una medida de probabilidad.Para cada $A\in \Omega$ defina:\,\\
\begin{equation*}
    \mathbb{P}(A)=\sum_{k\in A}\,\frac{2k}{n(n+1)}
\end{equation*}
\begin{equation*}
    \mathbb{P}(A)=\prod _{k\in A}\,\left(1-\frac{1}{k}\right)
\end{equation*}}
\end{tcolorbox}
\begin{proof}\,\\
    i)\,\,Afirmamos que $\mathbb{P}$ definida como:\,\\
    \begin{equation*}
        \mathbb{P}(A)=\sum_{k\in A}\,\frac{2k}{n(n+1)}\,\,\,(A\subseteq \Omega)
    \end{equation*}
    Es una medida de propabilidad en $\Omega$, para probarlo veremos que cumple con los tres axiomas:\,\\
    \,\\
    \begin{itemize}
        \item Tenmos que $\mathbb{P}(A)$ es positiva para tomo $A\subset \Omega$, pues para cualquier $x\in A$ $x>0$, ademas como $n>0$ se sigue que 
        $\frac{x}{n(n+1)}>0$ y por tanto\,\\
        \begin{equation*}
            \mathbb{P}(A)=\sum_{k\in A}\,\frac{2k}{n(n+1)}\geq 0
        \end{equation*}\,\\
        \item Tenemos que:\,\\
        \begin{equation*}
            \mathbb{P}(\Omega)=\sum_{k\in \Omega}\,\frac{2k}{n(n+1)}=\sum_{k\in \N_n/ \{0\}}\,\frac{2k}{n(n+1)}=\frac{2}{n(n+1)}\,\sum_{k\in\N_n/\{0\}}\,k=\frac{2}{n(n+1)}\frac{n(n+1)}{2}=1
        \end{equation*}
        \item Tomemos $\{A_k\}_{k\in\N}$ con los $A_K$ disjuntos a pares, entonces:\,\\
        \begin{equation*}
            \mathbb{P}\left(\bigcup_{k\in \N}\,A_K\right)=\sum_{a\in \bigcup_{k\in \N}\,A_K}\frac{2a}{n(n+1)}
        \end{equation*}\,\\
        Como todos los $A_K$ son disjuntos (no se suma dos veces un elemento) y usando la conmutatividad y asociatividad de la serie tenemos podemos reescribir esto coomo:\,\\
        \,\\
        \begin{equation*}
            \mathbb{P}\left(\bigcup_{k\in \N}\,A_K\right)=\sum_{a\in A_1}\frac{2a}{n(n+1)}+\sum_{a\in A_2}\frac{2a}{n(n+1)}+\cdots+\sum_{a\in A_K}\frac{2a}{n(n+1)}+\cdots=\sum_{k\in \N}\,\mathbb{P}(A_K)
        \end{equation*} 
    \end{itemize}\,\\
    Por tanto $\mathbb{P}$ es una medida de probabilidad en $\Omega$\,\\
    \,\\
    ii)\,\,Afirmamos que $\mathbb{P}$ definidad como:\,\\
    \,\\
    \begin{equation*}
         \mathbb{P}(A)=\prod _{k\in A}\,\left(1-\frac{1}{k}\right)\,\,\,(A\subseteq \Omega)
    \end{equation*}\,\\
    No es una medida de probabilidad en $\Omega$ pues:\,\\
    \,\\
    \begin{equation*}
         \mathbb{P}(\Omega)=\prod _{k\in \omega}\,\left(1-\frac{1}{k}\right)=\prod_{k\in \N_k/\{0\}}\,\left(1-\frac{1}{k}\right)=\left(1-\frac{1}{1}\right)\cdots\left(1-\frac{1}{n}\right)=0
    \end{equation*}\,\\
    Por tanto $\mathbb{P}$ no cumple el segundo axioma, es decir no es una medida de probabilidad en $\mathbb{P}$
    
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 3}},]
\textit{Un grupo de personas está compuesto de $60\%$  hombres y $40\%$  mujeres.
De los hombres $30\%$ fuma y de las mujeres el $20\%$ fuma. Si una persona
de este grupo se elige al azar. Determine las siguientes probabilidades:
\begin{itemize}
    \item Sea hombrey fuma
    \item Sea hombra y no fuma
    \item Sea mujer y fuma 
    \item Sea mujer y no fuma
    \item Sea hombre dado que se sabe que fuma
    \item Sea mujer dado que se sabe que no fuma
\end{itemize}
}
\end{tcolorbox}
\begin{proof}\,\\
    Sea \( H \) el evento "ser hombre", \( M \) el evento "ser mujer", \( F \) el evento "fumar".
    
    Sabemos:
    \begin{itemize}
        \item \( P(H) = 0.6 \), \( P(M) = 0.4 \)
        \item \( P(F|H) = 0.3 \), \( P(F|M) = 0.2 \)
    \end{itemize}
    
    Usamos la regla del producto: \( P(A \cap B) = P(A)P(B|A) \).
    
    \begin{enumerate}
        \item \( P(H \cap F) = 0.6 \times 0.3 = 0.18 \)
        \item \( P(H \cap F^c) = 0.6 \times 0.7 = 0.42 \)
        \item \( P(M \cap F) = 0.4 \times 0.2 = 0.08 \)
        \item \( P(M \cap F^c) = 0.4 \times 0.8 = 0.32 \)
        \item \( P(H|F) = \frac{P(H \cap F)}{P(F)} = \frac{0.18}{0.18 + 0.08} = \frac{9}{13} \)
        \item \( P(M|F^c) = \frac{P(M \cap F^c)}{P(F^c)} = \frac{0.32}{0.32 + 0.42} = \frac{8}{15} \)
    \end{enumerate}
\end{proof}\,\\

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 4}},]
\textit{Una caja contiene $n$ bolas rojas y $n$ bolas blancas. Se extraen dos bolas
al azar.
\begin{itemize}
    \item ¿Cu\'al es el espacio muestral para este experimento?
    \item ¿Cu\'al es la probabilidad de que las dos bolas tengan colores distintos?
    \item Encuentre la probabilidad $p_n$ de que las doa bolas sean del mismo color y obtenga $\displaystyle \lim_{n\rightarrow \infty}\,p_n$
\end{itemize}
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
    i)\,Definimos $R:=\{r_i|i\in \N_N\}$ que representa las $n$ pelotas rojas, de la misma manera Definimos
    $B:=\{b_i|i\in \N_n\}$ para las $n$ pelotas blancas, podemos decir que el espacio muestral del experimento es:\,\\
    \,\\
    \begin{equation*}
        \Omega:=\binom{R\cup B}{2}:=\{A\subset R\cup B\,\,|\,\,|A|=2\} 
    \end{equation*}\,\\
    Es decir la cantidad de subconjuntos de dos elementos de $R\cup B$, esto modela perfectamente el espacio muestral, pues nos muestra
    la cantidad de maneras en que podemos tomar solo dos bolas de las $2n$ posibles\,\\
    \,\\
    ii)\,\,Tenemos que podemos elegir $n$ bolas rojas diferentes, y $n$ bolas blnacas diferentes, por principio multiplicativo de conteo tenemos que
    podemos elegir $n^2$ maneras de elegir una roja y una blanca, por tanto denotemos a $D$ como el evento de sacr dos bolas y que sean de colores diferentes:\,\\
    \,\\
    \begin{equation*}
        P(A)=\frac{n^2}{\binom{2n }{2}}=\frac{n^2}{\frac{(2n)!}{2(2n-2)!}}=\frac{n^2}{\frac{(2n-1)2n}{2}}=\frac{n}{(2n-1)}
    \end{equation*}\,\\
    iii)\,\,Tenemos que el evento de conseguir dos bolas del mismo color en realidad es el complemento de $A$, por tanto aplicando la formula para la probabilidad del complemento:\,\\
    \,\\
    \begin{equation*}
        p_n=P(A^c)=1-P(A)=1-\frac{n}{2n-1}=\frac{n-1}{2n-1}=\frac{1-\frac{1}{n}}{2-\frac{1}{n}}
    \end{equation*}\,\\
    Finalmente tomamos el l\'imite al infinito de esta expresi\'on:\,\\
    \begin{equation*}
        \lim_{n\rightarrow \infty}\,\frac{1-\frac{1}{n}}{2-\frac{1}{n}}=\frac{1}{2}
    \end{equation*}
    
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 5}},]
\textit{Considere un experimento aleatorio cuyo espacio muestral es el conjunto
de pares de números $(x, y)$, donde $x$ y $y$ toman valores en $1,\cdots,n$, suponga
que cualquiera de estos puntos ocurre con idéntica probabilidad. Calcule
la probabilidad de que, al efectuar una vez el experimento aleatorio, se
obtenga un punto $(x, y)$:
\begin{itemize}
    \item Tal que $x=y$
    \item Tal que $x=1$ o $x=n$
    \item Tal que $x\leq y$
    \item Tal que $|x-y|\leq 1$
\end{itemize}}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
    Tenemos que la cantidad de elegir parejas ordenadas dentro de $\N_n$ es $n^2$, pues para elegir la primera entrada tenemos $n$ opciones y para la segunda
    tambi\'en tenemos $n$ opciones, por pirncipio multiplicativo de conteo tenemos $n^2$ formas diferentes de tener prejas ordenadas, ahora:\,\\
    \begin{enumerate}
        \item Si $x=y$ podemos armar $n$ posibles parejas donde ambas entradas son iguales por tanto la probabilidad de obtener una pareja ordenada es:\,\\
        \begin{equation*}
            \frac{n}{n^2}=\frac{1}{n}
        \end{equation*}
        \item Si $x=1$ tenemos $n$ formas de elegir la segunda entrada, analogamente si $x=n$ tenemos $n$ maneras diferentes de elegir la otra entrada, por tanto
        tenemos $2n$ formas de elegir parejas cuya primera entrada sea $n$ o $1$, por tanto la probabilidad de obtener una de estas parejas es:\,\\
        \begin{equation*}
            \frac{2n}{n^2}=\frac{2}{n}
        \end{equation*}
        \item Podemos elegir de $n$ formas la primera entrada, supongamos que fijamos $x=k$ con $1\leq k\leq n$, tenemos $n-k+1$ n\'umeros mayores iguales que $k$, por tanto la probabilidad de obtener una pareja 
        que cumpla con que la primera entrada es menor igual que la segunda:\,\\
        \,\\
        \begin{equation*}
            \frac{\displaystyle \sum_{k\in \N_n/\{0\}}n-k+1}{n^2}=\frac{n(n+1)-\frac{n(n+1)}{2}}{n^2}=\frac{n+1}{2n}
        \end{equation*}\,\\
        \item como $x,y\in \N$, si $|x-y|\leq 1$, entonces $x=y+1$ o $y=x+1$ o $x=y$, las formas de elegir parejas de naturales concecutivas es $n-1$, las contamos dos veces pues si $(x,y)$ es valida entonces $(y,x)$ taambi\'en es valida, Finalmente
        el total de parejas que podemos elegir tales que $x=y$ es $n$, entonces la probabilidad de encontar estas parejas:\,\\
        \,\\
        \begin{equation*}
            \frac{2(n-1)+n}{n^2}=\frac{3n-2}{n^2}
        \end{equation*}    
    \end{enumerate}
    
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 6}},]
\textit{Una urna contiene $3$ canicas rojas y $7$ negras, $A$ y $B$ juegan a sacar las
canicas y gana quien saque la primera roja, si $A$ saca la primer vez, ¿Cuál
es la probabilidad de que $A$ gane?. Responder para los casos con y sin
reemplazo.
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
    i)\,\,Caso sin reemplazo\,\\
    \,\\
    Tenemos que el sacar canicas en turnos concecutivos son eventos independientes, pues al haber reemplazo la probabilidad de uno no afecta la del otro, es decir, sacar una bola roja en el turno uno, sigue siendo igual de probable que
    sacarla en el turno $10000$, como todos los eventos son independientes entre si tenemos que la probabilidad de la intersecci\'on entre ellos es el producto de las probabilidades individuales, queremos encontrar la probabilidad de sacar una bola roja en un turno impar $k$, esto sera igual a:\,\\
    \,\\
    \begin{equation*}
        P\left(\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k \right)=\left(\prod_{i\in \N_{k-1}}\,P(B_i)\right)\,P(R_k)
    \end{equation*}\,\\
    Donde $B_i$ es sacar una canica blanca en el turno $i$-esimo, analogamente $R_k$ es sacar una canica roja en el turno $k$-esimo, finalmente valuando las probabilidades tenemos que:\,\\
    \,\\
    \begin{equation*}
        P\left(\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k \right)=\left(\frac{7}{10}\right)^{k-1}\,\frac{3}{10}
    \end{equation*}\,\\
    Tenemos que al haber reemplazo el juego puede prolongarse infinitamente, por tanto debemos sumar las probabilidades de siempre ganar en un turno impar:\,\\
    \,\\
    \begin{equation*}
        P\left(\bigcup_{k\in \N_p}\left(\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k\right) \right)
    \end{equation*}\,\\
    Donde $\N_p$ representa los naturales impares, luego tenemos que todos los eventos $\displaystyle\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k$ son ajenos entre si para cada todo $k\in \N_p$, pues si se obtiene una bola roja en el $k$-esimo turno impar el juego termina y no se avanzan más turnos\,\\
    \,\\
    \begin{equation*}
        P\left(\bigcup_{k\in \N_p}\left(\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k\right) \right)=\sum_{k\in \N_p}\,P\left(\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k\right)=\sum_{k\in \N_p}\,\left(\frac{7}{10}\right)^{k-1}\,\frac{3}{10}
    \end{equation*}\,\\
    \,\\
    Finalmente si $k$ es impar $k-1$ es par por tanto la suma puede verse como:\,\\
    \,\\
    \begin{equation*}
     P\left(\bigcup_{k\in \N_p}\left(\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k\right) \right)=\sum_{k\in \N}\,\left(\frac{7}{10}\right)^{2k}\,\frac{3}{10}=\frac{3}{10}\sum_{k\in \N}\,\left(\frac{49}{100}\right)^k=\frac{3}{10}\left(\frac{1}{1-\frac{49}{100}}\right)
    \end{equation*}\,\\
    \,\\
    Por tanto tenemos que:\,\\
    \begin{equation*}
      P\left(\bigcup_{k\in \N_p}\left(\left(\bigcap_{i\in \N_{k-1}}B_i\right)\cap R_k\right) \right)=\frac{3}{10}\,\left(\frac{100}{100-49}\right)=\frac{3}{10}\,\frac{100}{51}=\frac{30}{51}=\frac{10}{17}  
    \end{equation*}\,\\
    \,\\
    ii)\,\,Caso sin reemplazo\,\\
    \,\\
    Tenemos que como no hay reemplazo $A$ solo puede ganar en los turnos $1,3,5,7$ pues si se sacan $7$ bolas blancas seguidas el jugador $B$ gana en el turno $8$, como sabemos la probabilidad de que $A$ gane en el turno $1$ es:\,\\
    \,\\
    \begin{equation*}
        P(G_1)=\frac{3}{10}
    \end{equation*}\,\\
    Luego si queremos que $A$ gane en el tercer turno se tienen que sacar primero dos canicas blancas en los dos primeros turnos, y sacar una roja en el siguiente turno, denotaremos esto como $P(B_1\cap B_2\cap R_3)$, donde $B_i$ es sacar una canica blanca en el $i$-esimo turno
    y $R_i$ sacar una canica roja en el turno $i$-esimo, condicionando tenemos que:\,\\
    \begin{equation*}
        P(R_3|B_1\cap B_2)P(B_1\cup B_2)=P(R_3\cap B_1 \cap B_2)
    \end{equation*}\,\\
    Aplicando la regla del producto obtenemos que:\,\\
    \begin{equation*}
        P(B_1\cap B_2)=P(B_2|B_1)P(B_1)=\frac{2}{3}\left(\frac{7}{10}\right)=\frac{14}{30}
    \end{equation*}
    Por tanto la probabilidad de que $A$ gane en el turno $3$ es:\,\\
    \,\\
    \begin{equation*}
        P(G_3)= P(R_3|B_1\cap B_2)\,\left(\frac{14}{30}\right)=\frac{3}{8}\left(\frac{14}{30}\right)=\frac{21}{120}
    \end{equation*}\,\\
    Podemos aplicar un proceso an\'alogo para el caso en que $A$ gane en el $5$-to turno:\,\\
    \,\\
    \begin{equation*}
        P(G_5)=P(R_5\cap B_1\cap B_2\cap B_3\cap B_4)=P(R_5|B_1\cap B_2\cap B_3\cap B_4)\,P(B_1\cap B_2\cap B_3\cap B_4)
    \end{equation*}\,\\
    Aplicando nuevamente la regla del producto:\,\\
    \begin{equation*}
        P(B_1\cap B_2\cap B_3\cap B_4)=P(B_4|B_1\cap B_2\cap B_3)\,P(B_3|B_1\cap B_2)\,P(B_1\cap B_2)
    \end{equation*}\,\\
    Valuando:\,\\
    \,\\
    \begin{equation*}
        P(B_1\cap B_2\cap B_3\cap B_4)=\left(\frac{4}{7}\right)\,\left(\frac{5}{8}\right)\,\left(\frac{7}{15}\right)=\frac{1}{6}
    \end{equation*}\,\\
    Por tanto:\,\\
    \,\\
    \begin{equation*}
        P(G_5)=P(R_5\cap B_1\cap B_2\cap B_3\cap B_4)=P(R_5|B_1\cap B_2\cap B_3\cap B_4)\,\left(\frac{1}{6}\right)=\frac{3}{6}\,\left(\frac{1}{6}\right)=\frac{1}{12}
    \end{equation*}\,\\
    Finalmente aplicando el proceso para que $A$ gane en el $7$-mo turno:\,\\
    \,\\
    \begin{equation*}
        P(G_7)=P(R_7\cap B_1\cap B_2\cap B_3\cap B_4\cap B_5 \cap B_6)=P(R_7|B_1\cap B_2\cap B_3\cap B_4\cap B_5 \cap B_6)\,P(B_1\cap B_2\cap B_3\cap B_4\cap B_5 \cap B_6)
    \end{equation*}\,\\
    Usando la regla del producto:\,\\
    \,\\
    \begin{align*}
        P(B_1\cap B_2\cap B_3\cap B_4\cap& B_5 \cap B_6)\,\\
        \,\\
        &=P(B_6|B_1\cap B_2\cap B_3\cap B_4\cap B_5)P(B_5|B_1\cap B_2\cap B_3\cap B_4)P(B_1\cap B_2\cap B_3\cap B_4)
    \end{align*}\,\\
    Valuando:\,\\
    \begin{equation*}
        P(B_1\cap B_2\cap B_3\cap B_4\cap B_5 \cap B_6)=\left(\frac{2}{5}\right)\,\left(\frac{1}{2}\right)\,\frac{1}{6}=\frac{1}{30}
    \end{equation*}
    Por tanto:\,\\
    \,\\
    \begin{equation*}
        P(G_7)=P(R_7\cap B_1\cap B_2\cap B_3\cap B_4\cap B_5 \cap B_6)=\left(\frac{3}{4}\right)\,\left(\frac{1}{30}\right)=\frac{1}{40}
    \end{equation*}\,\\+
    Concluimos que la probabilidad de que $A$ gane el juego es:\,\\
    \,\\
    \begin{equation*}
        P(G_A)=P(G_1)+P(G_3)+P(G_5)+P(G_7)=\frac{3}{10}+\frac{21}{120}+\frac{1}{12}+\frac{1}{40}=\frac{7}{12}
    \end{equation*}

\end{proof}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 7}},]
\textit{Una moneda balanceada es lanzada 5 veces. En cada lanzamiento se coloca
en una urna una bola blanca cuando sale águila y una roja si sale sol. En
seguida se sacan sin reemplazo $n (n = 1, . . . , 5)$ bolas de la urna. Calcule la
probabilidad de que la urna contenga sólo bolas blancas bajo la hipótesis
de que las $n$ bolas extraídas son blancas.
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
Particionemos el espacio de probabilidad en los siguientes eventos $\{R_i\}_{i\in \N_5}$, donde $R_i$ es que haya $i$ bolas rojas en la urna, supongamos que extraemos $k$ bolas blancas de la urna con $k\in \{1,2,3,4,5\}$,
Se nos pide encontrar la probabilidad $P(R_0|B_k)$, es decir la probabilidad de que haya $0$ bolas rojas dado que extraemos $k$ blancas,
Notemos que para todo $k\in\{1,2,3,4,5\}$ $P(B_k|R_0)=1$, pues todas las bolas son blancas, mientras tanto tenemos que para $j\in\N_5$:\,\\
\,\\
\begin{equation*}
    P(R_j)=\frac{\binom{5}{j}}{2^5}\neq 0
\end{equation*}\,\\
Pues por cada tiro tenemos dos opciones (aguila-sol), si hacemos $5$ tiros hay $2^5$ posibles combinaciones de resultados, mientras que $\binom{5}{j}$ son las formas en que pueden salir $j$ resultados iguales, ahora Notemos
que si $j>5-k$, entonces $P(B_k|R_j)=0$, pues $k+j>5$ lo que implica que es imposible armar subconjuntos con $k$ pelotas, pues estamos suponiendo que hay al menos $j$ rojas, finalmente afirmamos que si $0\leq j \leq 5-k$ entonces:\,\\
\,\\
\begin{equation*}
    P(B_k|R_j)=\frac{\binom{5-j}{k}}{\binom{5}{k}}
\end{equation*}\,\\
Pues queremos la cantidad de subconjuntos de $k$ elementos dado que todas las pelotas son blancas por eso quitamos las $j$ rojas, sobre el total de subconjuntos de $k$ elementos que se le puede sacar normalmente a la urna, ademas con esta suposicii\'on $P(B_k)\neq 0$ aplicando el teorema de Bayes:\,\\
\,\\
\begin{equation*}
    P(R_0|B_k)=\frac{P(B_k|R_0)\,P(R_0)}{\sum_{j\in \N_5}P(B_k|R_j)P(R_j)}
\end{equation*}\,\\
Finalmente aplicando todas las consideraciones:\,\\
\,\\
\begin{equation*}
    P(R_0|B_k)=\frac{\displaystyle\binom{5}{k}}{\displaystyle\sum_{j=0}^{5-k}\,\binom{5-j}{k}\,\binom{5}{j}}
\end{equation*}\,\\
Valuando:\,\\
\,\
\begin{equation*}
    P(R_0|B_5)=\frac{\displaystyle \binom{5}{5}}{\displaystyle\sum_{j=0}^{0}\,\binom{5-j}{5}\,\binom{5}{j}}=1
\end{equation*}\,\\
\begin{equation*}
    P(R_0|B_4)=\frac{\displaystyle\binom{5}{4}}{\displaystyle\sum_{j=0}^{1}\,\binom{5-j}{4}\,\binom{5}{j}}=\frac{1}{2}
\end{equation*}\,\\
\begin{equation*}
    P(R_0|B_3)=\frac{\displaystyle\binom{5}{3}}{\displaystyle\sum_{j=0}^{2}\,\binom{5-j}{3}\,\binom{5}{j}}=\frac{1}{4}
\end{equation*}\,\\
\begin{equation*}
    P(R_0|B_2)=\frac{\displaystyle\binom{5}{2}}{\displaystyle\sum_{j=0}^{3}\,\binom{5-j}{2}\,\binom{5}{j}}=\frac{1}{8}
\end{equation*}\,\\
\begin{equation*}
    P(R_0|B_1)=\frac{\displaystyle\binom{5}{1}}{\displaystyle\sum_{j=0}^{4}\,\binom{5-j}{2}\,\binom{5}{j}}=\frac{1}{16}
\end{equation*}\,\\
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 8}},
]
\textit{Tres sucursales de una tienda tienen $8$, $12$ y $14$ empleados de los cuales $4$,
$7$ y $10$ son mujeres, respectivamente.
\begin{itemize}
    \item Se escoge una sucursal al azar y de ella se escoge un empleado. Si este es mujer, ¿cuál es la
probabilidad de que trabaje en la sucursal con $12$ empleados?
\item Si se escoge un segundo empleado de la misma sucursal, ¿cuál es la
probabilidad de que se escoja una mujer?
\end{itemize}
}
\end{tcolorbox}

\begin{proof}\,\\
	\textbf{Notación:}
	\begin{itemize}
	    \item Sea $S_1$, $S_2$, $S_3$ los eventos de elegir la sucursal con 8, 12 y 14 empleados respectivamente.
	    \item Sea $M$ el evento el empleado elegido es mujer.
	\end{itemize}
	
	\textbf{Datos:}
	\begin{itemize}
	    \item $P(S_1) = P(S_2) = P(S_3) = \frac{1}{3}$ (ya que la sucursal se elige al azar).
	    \item $P(M|S_1) = \frac{4}{8} = \frac{1}{2}$,\quad
	          $P(M|S_2) = \frac{7}{12}$,\quad
	          $P(M|S_3) = \frac{10}{14} = \frac{5}{7}$
	\end{itemize}

	\vspace{0.2cm}
	\textbf{Inciso 1.} Queremos calcular \( P(S_2 | M) \), es decir, la probabilidad de que la persona elegida trabaje en la sucursal con 12 empleados, dado que fue mujer. Aplicamos el teorema de Bayes:

	\[
	P(S_2 | M) = \frac{P(M|S_2) P(S_2)}{P(M)}
	\]

	Donde:
	\[
	P(M) = \sum_{i=1}^3 P(M|S_i)P(S_i) 
	= \frac{1}{3}\left(\frac{1}{2} + \frac{7}{12} + \frac{5}{7}\right)
	\]

	Calculamos:
	\[
	\frac{1}{2} = \frac{42}{84},\quad
	\frac{7}{12} = \frac{49}{84},\quad
	\frac{5}{7} = \frac{60}{84}
	\Rightarrow
	P(M) = \frac{1}{3} \left( \frac{42 + 49 + 60}{84} \right) = \frac{1}{3} \cdot \frac{151}{84} = \frac{151}{252}
	\]

	Entonces:
	\[
	P(S_2 | M) = \frac{\frac{7}{12} \cdot \frac{1}{3}}{\frac{151}{252}} 
	= \frac{7}{36} \cdot \frac{252}{151}
	= \frac{7 \cdot 252}{36 \cdot 151}
	= \frac{49}{151}
	\]

	\vspace{0.2cm}
	\textbf{Respuesta:} La probabilidad de que la mujer seleccionada trabaje en la sucursal de 12 empleados es $\frac{49}{151}$

	\vspace{0.4cm}
	\textbf{Inciso 2.} Ahora, si sabemos que se eligió una sucursal al azar, un primer empleado (que fue mujer), y se selecciona un segundo empleado de la misma sucursal, queremos saber la probabilidad de que ese segundo empleado también sea mujer.

	Usamos la ley de la probabilidad total condicional:

	\[
	P(\text{2da mujer}) = 
	\sum_{i=1}^{3} P(\text{2da mujer} | M \cap S_i) \cdot P(S_i | M)
	\]

	Pero en realidad, dado que el primer empleado es mujer, quedan menos mujeres disponibles en la sucursal seleccionada. Entonces:

	\begin{itemize}
	    \item En \( S_1 \): $3$ mujeres de $7$ empleados restantes → \( \frac{3}{7} \)
	    \item En \( S_2 \): $6$ mujeres de $11$ empleados restantes → \( \frac{6}{11} \)
	    \item En \( S_3 \): $9$ mujeres de $13$ empleados restantes → \( \frac{9}{13} \)
	\end{itemize}

	Usamos los valores de \( P(S_i|M) \) previamente calculados:

	\[
	P(S_1|M) = \frac{\frac{1}{2} \cdot \frac{1}{3}}{\frac{151}{252}} = \frac{42}{151},\quad
	P(S_2|M) = \frac{49}{151},\quad
	P(S_3|M) = \frac{60}{151}
	\]

	Entonces:

	\[
	P(\text{2da mujer}) = \frac{3}{7} \cdot \frac{42}{151} + \frac{6}{11} \cdot \frac{49}{151} + \frac{9}{13} \cdot \frac{60}{151}
	\]

	Calculamos:

	\[
	= \frac{18}{151} + \frac{294}{11 \times 151} + \frac{540}{13 \times 151}
	= \frac{1}{151} \times (18 +\frac{294}{11} + \frac{540}{13})
        = \frac{2574}{143} + \frac{3822}{143} + \frac{5940}{143} = \frac{12336}{151 \times 143} ? \frac{12336}{21593}
	\]

	Por tanto, la probabilidad de que el segundo empleado también sea mujer es aproximadamente {0.5712}, o se puede dejar expresado como fracción si se desea exactitud.
\end{proof}
