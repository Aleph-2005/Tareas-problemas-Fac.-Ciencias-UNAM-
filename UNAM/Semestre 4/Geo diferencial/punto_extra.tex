\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Superficie Regular}},]
\textit{Sea $S\subset \R^3$, se dice que $S$ es regular si cumple con las siguientes condiciones:\,\\
\begin{enumerate}
    \item Para todo $\alpha\in S$ existe $V\subset \R^3$ vecindad abierta, $U\subset \R^2$ abierto y $F:U\to S\cap V$ tal que $\alpha\in V$ y $F$ es un homeomorfismo
    \item La matriz jacobiana $J$ de $F$ en el punto $\alpha$ tiene rango completo
\end{enumerate}}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Topologia de superficie regular}},]
\textit{Decimos que $A\subseteq S$ es abierto si y solo si para cualquier $F:U\to S\cap V$ parametrizaci\'on local de una vecindad coordenada  se tiene que $F^{-1}[S\cap V\cap A]\subset U$ es abierto en $\R^2$}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Topologia inducida }},]
\textit{Sea $S\subset \R^3$ y $A\subseteq S$, $A$ es abierto en $S$ si existe $V_A\subseteq \R^3$ abierto en $\R^3$ tal que $A\subseteq V_A$ y $A=V_A\cap S$}
\end{tcolorbox}


\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema de punto extra}},]
\textit{Demuestre que la topologia inducida coincide con la topologia de superficie regular}
\end{tcolorbox}
\begin{proof}
    Sea $A\subseteq S$ demostraremos que $A$ es abierto bajo la topologia de superficie regular si y solo si es abierto bajo la topologia inducida:\,\\
    \,\\
    $\Leftarrow)$ Primero supongamos que $A\subseteq S$ es abierto en la topologia de superficie regular, sea $p\in A\subseteq S$ existe $U_p\subset \R^2$ abierto, $V_p\subset R^3$ abierto que cumple con $p\in V_p$ y $F_p:U_p\to S\cap V_p$ parametrizaci\'on local, como $A\subseteq S$ es abierto bajo la topologia 
    de superficie regular entonces $U=F^{-1}[S\cap V_P\cap A]\subset \R^2$ es abierto, como $F_p$ es parametrizacion local tenemos que $F_p$ es biyectiva con  inversa continua, por la caracterizaci\'on topologica de la continuidad aplicada a $F^{-1}_p$ existe $\bar{V}\subset \R^3$ abierto tal que $S\cap V_p\cap A=F_p[U]=S\cap V_p\cap \bar{V}_p$, como hemos visto para cada $p\in A$ podemos construir $\bar{V}_p\cap V_p\subset \R^3$ abierto el cual cumple
    con $p\in \bar{V_p}\cap V_p$, pues $p\in S\cap A\cap V_p =S\cap V_p\cap \bar{V_P}$, por tanto $V=\displaystyle\bigcup_{p\in A}\bar{V_p}\cap V_p\subset \R^3$ es abierto  que cumple con $A\subset V$ ademas que $S\cap V=S\cap\left(\displaystyle\bigcup_{p\in A}\bar{V}_p\cap V_p\right)=\displaystyle\bigcup_{p\in A}(S\cap \bar{V}_p\cap V_p)=\bigcup_{p\in A}(S\cap A\cap V_p)=\left(S\cap\bigcup_{p\in A}V_p\right)\cap A$,
    finalmente como $A\subseteq S$ y $A\subseteq \displaystyle\bigcup_{p\in A}\,V_p$ se sique que:\,\\
    \begin{equation*}
        S\cap V=\left(S\cap\bigcup_{p\in A}V_p\right)\cap A=A
    \end{equation*}
    Por tanto $A$ es abierto en la topologia inducida por $S$\,\\
    \,\\
    $\Rightarrow)$ Supongamos que $A\subseteq S$ es abierto en la topologia inducida de $S$, por tanto existe $V_A\subset \R^3$ abierto tal que $A=S\cap V_A$, sea $F:U\to S\cap V$ una vecindad coordenada de $S$, tenemos que
    $A\cap V\cap S=S\cap(V\cap V_A)$, ahora tenemos que como $F$ es continua, y $V_A\subset \R^3$ es abierto tenemos que existe $U_a\subset \R^2$ abierto tal que $F^{-1}[V_a]=U\cap U_a$, ahora tenemos que:
    \begin{equation*}
        F^{-1}[(S\cap V)\cap V_A]=F^{-1}[(S\cap V)]\cap F^{-1}[V_A]=U\cap(U\cap U_A)=U\cap U_A\subset \R^2
    \end{equation*}
    Por tanto $ F^{-1}[S\cap V\cap A]$ es abierto.
\end{proof}
