\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 1}},]
\textit{Dado un espacio de probabilidad $(\Omega,\mathfrak{F}  ,\mathbb{P})$ en donde
  $\Omega=\{a,b,c\}$ y $\mathfrak{F} =\{\emptyset,\Omega,\{a\},\{b,c\}\}$,
  considere $(\R,\mathfrak{B} )$ el conjunto de n\'umeros reales con la
  $\sigma$-\'algebra de Borel y la funci\'on $X:\Omega\to\R$ definida
  por $X(a)=1$,$X(b)=2$ y $X(c)=3$. ¿Es $X$ una v.a?}
\end{tcolorbox}
\begin{proof}\,\\
  Tomemos $I:=(-\infty,2]$, un subconjunto de Borel en $\R$, notemos que:\,\\
  \begin{equation*}
    X^{-1}[I]:=\{a,b\}\notin \mathfrak{F} 
  \end{equation*}\,\\
  Por tanto $X$ no es variable aleatoria
\end{proof}\,\\
\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 2}},]
\textit{Para las siguientes funciones  encuentra $\theta$ que las haga funciones de densidad y calcula su respectiva funci\'on de distribuci\'on.\,\\
\,\\
\begin{equation*}
    a)f_X(x)=\begin{cases} 
      \frac{\theta}{2^x}              & \mbox{si } x=1,2,\dots   \\
      0 \cdot                         & \text{otro caso}
   \end{cases}
\end{equation*}\,\\
\,\\
\begin{equation*}
    b)f_X(x)=\begin{cases} 
      \theta x^2             & \mbox{si } -1\leq x\leq 1  \\
      0 \cdot                         & \text{otro caso}
   \end{cases}
\end{equation*}\,\\
\,\\
Y adem\'as calcula para el inciso a) $\mathbb{P}[X<5]$, $\mathbb{P}[X=10]$, $\mathbb{P}[X\geq 4]$, $\mathbb{P}[6\leq X<10]$ y para el inciso b) $\mathbb{P}[|X|<1/2]$, $\mathbb{P}[-3/4<X<1/2]$, $\mathbb{P}[X\geq 1/4]$, $\mathbb{P}[X>0]$.
}
\end{tcolorbox}
\begin{proof}\,\\
  \textbf{a)}\,\,Para que $f_X(x)$ sea funci\'on de densidad debe de cumplir que:\,\\
  \,\\
  \begin{equation*}
    \sum_{i=1}^{\infty}\,f_X(i)=1
  \end{equation*}\,\\
  Pues $f_X(x)$ solo toma valores en enteros positivos, reescribiendo la condici\'on:\,\\
  \,\\
  \begin{equation*}
    \sum_{i=1}^{\infty}\,f_X(i)=\sum_{i=1}^{\infty}\,\frac{\theta}{2^i}=\frac{\theta}{2}\,\sum_{i=1}^{\infty}\,\frac{1}{2^{i-1}}=1
  \end{equation*}\,\\
  Reducuiendo la suma geometrica obtenemos:\,\\
  \begin{equation*}
    \frac{\theta}{2}\,\left(\frac{1}{1-\left(\frac{1}{2}\right)}\right)=1\implies 2\,\,\frac{\theta}{2}=1\implies \theta=1
  \end{equation*}\,\\
  Finalmente si $\theta=1$ tenemos que $f_X(i)=\frac{1}{2^i}\geq 0$ para todo $i\in \Z^{+}$, por tanto:\,\\
  \begin{equation*}
    a)f_X(x)=\begin{cases} 
      \frac{1}{2^x}              & \mbox{si } x=1,2,\dots   \\
      0 \cdot                         & \text{otro caso}
   \end{cases}
\end{equation*}\,\\
Es una funci\'on de densidad discreta.\,\\
\,\\
Recordemos que por ser una variable discreta la funci\'on de distribuci\'on se define para $x\in \R$ como:\,\\
\begin{equation*}
  F(x)=\sum_{u\leq x}\,f_X(u)
\end{equation*}\,\\
En particulas si $x\in \Z^{+}$ obtendremos la suma parcial de una serie geometrica:\,\\
\begin{equation*}
  F(x)=\sum_{u\leq x}\,f_X(u)=\sum_{i=1}^{x}\,\frac{1}{2^x}=\frac{1}{2}\,\sum_{i=1}^{x}\,\frac{1}{2^{i-1}}=\frac{1}{2}\sum_{l=0}^{x-1}\frac{1}{2^l}=\frac{1}{2}\,\left(\frac{1-\frac{1}{2^x}}{1-\frac{1}{2}}\right)=1-\frac{1}{2^x}
\end{equation*}\,\\
Por tanto la funci\'on de distribuci\'on seria:\,\\
\begin{equation*}
    a)F(x)=\sum_{u\leq x}\,f_X(u)=\begin{cases} 
      1-\frac{1}{2^\lfloor x\rfloor }             & \mbox{si } 1\leq x  \\
      0 \cdot                         & \text{otro caso}
   \end{cases}
\end{equation*}\,\\
Calculando:\,\\
\begin{itemize}
  \item $P[X<5]$:\,\\
  \begin{equation*}
    \sum_{i=1}^{4}\,\frac{1}{2^i}=1-\frac{1}{2^4}=\frac{15}{16}
  \end{equation*}\,\\
  \item $P[X=10]$\,\\
  \begin{equation*}
    f_X(10)=\frac{1}{2^{10}}
  \end{equation*}\,\\
  \item $P[X\geq 4]$:\,\\
  \begin{equation*}
    P[X\geq 4]=1-P[X<4]=1-\sum_{i=1}^{3}\frac{1}{2^i}=\frac{1}{8}
  \end{equation*}\,\\
  \item $P[6\leq X<10]$:\,\\
  \begin{equation*}
    \sum_{i=6}^{9}\,\frac{1}{2^i}=\frac{15}{512}
  \end{equation*}
\end{itemize}\,\\
\textbf{b)}\,Para que $f_X$ sea uina funci\'on de densidad debe cumplirse que:\,\\
\begin{equation*}
  \int_{-\infty}^{\infty}\,f_X(x)=\int_{-1}^{1}\,\theta\,x^2=1
\end{equation*}\,\\
Valuando la integral:\,\\
\begin{equation*}
  \theta\,\int_{-1}^{1}\,x^2=\theta\,\frac{x^3}{3}\big|_{x = -1}^{x = 1}=\theta\,\frac{2}{3}=1\implies \theta=\frac{3}{2}
\end{equation*}\,\\
Como $f_X(x)=\frac{3}{2}\,x^2\geq 0$ para toda $x\in [-1,1]$, por tanto:\,\\
\begin{equation*}
   b)f_X(x)=\begin{cases} 
      \frac{3}{2}\,x^2             & \mbox{si } -1\leq x\leq 1  \\
      0 \cdot                         & \text{otro caso}
   \end{cases}
\end{equation*}\,\\
Es una funci\'on de densidad luego su funci\'on de distribuci\'on esta dada por:\,\\
\begin{equation*}
  F(x)=\int_{-\infty}^{x}\,f(u)\,du
\end{equation*}\,\\
Si tomamos $x\in[-1,1]$ tenemos que:\,\\
\begin{equation*}
  F(x)=\int_{-\infty}^{x}\,f(u)\,du=\int_{-\infty}^{x}\,\frac{3}{2}\,x^2\,dx=\frac{3}{2}\,\frac{t^3}{3}\big|_{t = -1}^{t = x}=\frac{3}{2}\,\left(\frac{x^3-1}{3}\right)
\end{equation*}\,\\
Por tanto la funci\'on de distribuci\'on es:\,\\
\begin{equation*}
   b)F(x)=\int_{-\infty}^{x}\,f(u)\,du=\begin{cases} 
      \frac{x^3-1}{3}        & \mbox{si } -1\leq x\leq 1  \\
      0                         & \mbox{si} x<-1 \\
      1                         &\text{en otro caso}
   \end{cases}
\end{equation*}\,\\
\begin{enumerate}
  \item $P[|X|<\frac{1}{2}]$:\,\\
  \begin{equation*}
      \int_{-\frac{1}{2}}^{\frac{1}{2}}\,\frac{3}{2}\,x^2=\frac{3}{2}\,\frac{x^3}{3}\big|_{x = -\frac{1}{2}}^{t = \frac{1}{2}}=\frac{1}{8}
  \end{equation*}\,\\
  \item $P[-\frac{3}{4}<X<\frac{1}{2}]$:\,\\
  \begin{equation*}
      \int_{-\frac{3}{4}}^{\frac{1}{2}}\,\frac{3}{2}\,x^2=\frac{3}{2}\,\frac{x^3}{3}\big|_{x = -\frac{3}{4}}^{t = \frac{1}{2}}=\frac{35}{128}
  \end{equation*}\,\\
  \item $P[X\geq \frac{1}{4}]$:
  \begin{equation*}
    \int_{\frac{1}{4}}^{1}\,\frac{3}{2}\,x^2=\frac{3}{2}\,\frac{x^3}{3}\big|_{x =\frac{1}{4}}^{t = 1}=\frac{63}{128}
  \end{equation*}\,\\
  \item $P[X>0]$:
  \begin{equation*}
    \int_{0}^{1}\,\frac{3}{2}\,x^2=\frac{3}{2}\,\frac{x^3}{3}\big|_{x=0}^{x= 1}=\frac{1}{2}
  \end{equation*}
\end{enumerate}
\end{proof}
\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 3}},]
\textit{Sean $F$ y $G$ dos funciones de distribuci\'on determine si las siguientes funciones son de distribuci\'on:
\begin{enumerate}
\item $aF+(1-a)G$ con $a\in[0,1]$,
\item $F+G$,
\item $\max(F,G)$.
\end{enumerate}}
\end{tcolorbox}
\begin{proof}\,\\
  \textbf{I)}\,\,Demostraremos que $R(x)=aF(x)+(1-a)G(x)$ es monotona sean $x_1,x_2\in \R$ tales que $x_1<x_2$, como $F$ y $G$ son funciones de 
  distribuci\'on se cumple que:\,\\
  \begin{equation*}
    F(x_1)<F(x_2)\implies a\,F(x_1)<a\,F(x_2)\,\,(a\in[0,1])
  \end{equation*}\,\\
  \begin{equation*}
    G(x_1)<G(x_2)\implies (1-a)\,G(x_1)<(1-a)G(x_2)\,\,(a\in [0,1])
  \end{equation*}\,\\
  Por tanto tenemos que:\,\\
  \begin{equation*}
    R(x_1)=a\,F(x_1)+(1-a)\,G(x_1)<a\,F(x_2)+(1-a)\,G(x_2)=R(x_2)
  \end{equation*}\,\\
  Ahora veamos que:\,\\
  \begin{equation*}
    \lim_{x\rightarrow \infty}\,aF(x)+(1-a)G(x)=a\,\lim_{x\rightarrow \infty}F(x)+\lim_{x\rightarrow \infty}\,G(x)-a\lim_{x\rightarrow \infty}\,F(x)=a-a+1=1
  \end{equation*}\,\\
  \begin{equation*}
    \lim_{x\rightarrow -\infty}\,aF(x)+(1-a)G(x)=\lim_{x\rightarrow -\infty}F(x)+(1-a)\lim_{x\rightarrow -\infty}\,G(x)=a\,0+(1-a)\,0=0
  \end{equation*}\,\\

  Como $F(x)$ y $G(x)$ son continuas por la derecha cualquier combinaci\'on lineal de ellas lo es, en particular $R(x)=aF(x)+(1-a)G(x)$ lo es, por tanto $R(x)$ es
  funci\'on de distribuci\'on\,\\
  \,\\
  \textbf{II)}\,\,Veamos que:\,\\
  \begin{equation*}
    \lim_{x\rightarrow \infty}\,F(x)+G(x)=\lim_{x\rightarrow \infty}F(x)+\lim_{x\rightarrow \infty}G(x)=1+1=2\neq 1
  \end{equation*}\,\\
  Por tanto $R(x)=G(x)+F(x)$ no es funci\'on de distribuci\'on\,\\
  \,\\
  \textbf{III)}\,\,Demostraremos que $R(x)=max(G(x),F(x))$ es monotona, tomemos $x_1,x_2\in \R$ tales que $x_1<x_2$, sea $l=max(G(x_2),F(x-2))$ tenemos que $F(x_1)<F(x_2)<l$ y $G(x_1)<G(x_2)<l$, por tanto $max\{F(x_1),G(x_1)\}<l$, es decir 
  $R(x)$ es mon\'otona\,\\
  \,\\
  Luego tenemos que:\,\\
  \begin{equation*}
    \lim_{x\rightarrow \infty}\,max\{G(x),F(x)\}=max\{\lim_{x\rightarrow \infty}\,G(x),\lim_{x\rightarrow \infty}\,F(x)\}=max\{1,1\}=1
  \end{equation*}\,\\
    \begin{equation*}
    \lim_{x\rightarrow -\infty}\,max\{G(x),F(x)\}=max\{\lim_{x\rightarrow -\infty}\,G(x),\lim_{x\rightarrow -\infty}\,F(x)\}=max\{0,0\}=0
  \end{equation*}\,\\
  Finalmente tenemos que si $G(x)$ y $F(x)$ son continuas por la derecha entonces $R(x)=max\{F(x),G(x)\}$ tambi\'en lo es, por tanto
  se sigue que $R(x)$ es funci\'on de distribuci\'on
\end{proof}\,\\

\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 5}},]
\textit{Un vendedor de enciclopedias visita dos casas y vende con probabilidad .6 la versi\'on est\'andar y .3 la extendida, si la est\'andar cuesta \$500 y la extendida, \$1000. Define la v.a. X como el total de las ventas. ?`Qu\'e valores puede tomar X y con qu\'e probabilidad?
}
\end{tcolorbox}
\begin{proof}\,\\
    
Notese $X \in \{0,\ 500,\ 1000,\ 1500,\ 2000\}$

\begin{itemize}
    \item \( X = 0 \):  
    \begin{itemize}
        \item (0, 0): \( 0.1 \cdot 0.1 = 0.01 \)
    \end{itemize}

    \item \( X = 500 \):  
    \begin{itemize}
        \item (0, 500): \( 0.1 \cdot 0.6 = 0.06 \)
        \item (500, 0): \( 0.6 \cdot 0.1 = 0.06 \)
        \item Total: \( P(X = 500) = 0.06 + 0.06 = 0.12 \)
    \end{itemize}

    \item \( X = 1000 \):  
    \begin{itemize}
        \item (0, 1000): \( 0.1 \cdot 0.3 = 0.03 \)
        \item (500, 500): \( 0.6 \cdot 0.6 = 0.36 \)
        \item (1000, 0): \( 0.3 \cdot 0.1 = 0.03 \)
        \item Total: \( P(X = 1000) = 0.03 + 0.36 + 0.03 = 0.42 \)
    \end{itemize}

    \item \( X = 1500 \):  
    \begin{itemize}
        \item (500, 1000): \( 0.6 \cdot 0.3 = 0.18 \)
        \item (1000, 500): \( 0.3 \cdot 0.6 = 0.18 \)
        \item Total: \( P(X = 1500) = 0.18 + 0.18 = 0.36 \)
    \end{itemize}

    \item \( X = 2000 \):  
    \begin{itemize}
        \item (1000, 1000): \( 0.3 \cdot 0.3 = 0.09 \)
    \end{itemize}
\end{itemize}


Valores de probabilidad de que \( X \) tome valores $\{0,500,1500,2000\}$\,\\
\begin{itemize}
    \item $P[X=0] =0.01$
    \item $P[X=500]= 0.12$
    \item $P[X=1000] = 0.42$
    \item $P[X=1500] = 0.36$
    \item $P[X=2000] = 0.09$ 
\end{itemize}

\end{proof}\,\\
\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 6}},]
\textit{Una refaccionar\'ia ha descubierto que el n\'umero $X$ de neum\'aticos que vende por semana sigue una distribuci\'on $Poisson$ con par\'ametro 10. Se sabe que la ganancia por cada neum\'atico vendido es de $200$ pesos. Sin embargo, un d\'ia al inicio de la semana los due\~nos de la refaccionar\'ia se encuentran con que s\'olo les quedan 10 neum\'aticos, y que lamentablemente durante esa semana no van a poder traer m\'as de su almac\'en. Determine la distribuci\'on que siguen las ganancias totales con respecto a los neum\'aticos a lo largo de esa semana.
}
\end{tcolorbox}
\begin{proof}
    Sea \( X \sim \text{Poisson}(10) \) la variable aleatoria que representa el número de neumáticos demandados en una semana. La ganancia por cada neumático vendido es de 200 pesos.

La ganancia total es:

\[
G = 200 \cdot Y
\]

Entonces, \( G \) toma valores en \( \{0, 200, 400, \dots, 2000\} \) ya que Y solo tenemos 10 neumáticos y su distribución es:

\[
P(G = 200k) =
\begin{cases}
\dfrac{10^k e^{-10}}{k!}, & \text{si } k = 0, 1, \dots, 9 \\
1 - \sum_{i=0}^{9} \dfrac{10^i e^{-10}}{i!}, & \text{si } k = 10
\end{cases}
\]
\end{proof}\,\\

\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 7}},]
\textit{Sea $X$ una v.a. con distribuci\'on de Poisson de par\'ametro $\lambda$. ?`Cu\'al es la probabilidad de que $X$ tome valor par (considere al cero como par)?
}
\end{tcolorbox}
\begin{proof}\,\\
  Tenemos que la funci\'on de densidad para $X$ es:\,\\
\begin{equation*}
P[X = k] =
\begin{cases}
\displaystyle \frac{\lambda^k e^{-\lambda}}{k!}, & \text{si } k = 0, 1, 2, \dots \\
0, & \text{en otro caso}
\end{cases}
\end{equation*}\,\\
Con $\lambda>0$, si queremos obtener la probabilidad de tomar un valor entero par, tenemos que:\,\\
\begin{equation*}
  P[X \equiv 0 \pmod{2}]=\sum_{k=0}^{\infty}\,\frac{\lambda^{2k}\,e^{-\lambda}}{(2K)!}=e^{-\lambda}\,\sum_{k=0}^{\infty}\,\frac{\lambda^{2k}}{(2K)!}
\end{equation*}\,\\
Recordando la serie de taylor para las funciones $f(x)=e^x$, $f(x)=e^{-x}$:\,\\
\begin{equation*}
  e^{x}=\sum_{i=0}^{\infty}\,\frac{x^i}{i!}
\end{equation*}\,\\
\begin{equation*}
  e^{-x}=\sum_{i=0}^{\infty}\,\frac{(-1)^i\,x^i}{i!}
\end{equation*}\,\\
Por tanto la serie de taylor de $f(x)=e^x+e^{-x}$:\,\\
\begin{equation*}
  e^{-x}+e^x=\sum_{i=0}^{\infty}\,\frac{x^i+(-x)^i}{i!}=2\,\sum_{l=0}^{\infty}\,\frac{x^{2l}}{(2l)!}
\end{equation*}\,\\
Reemplazando:\,\\
\begin{equation*}
  P[X \equiv 0 \pmod{2}]=e^{-\lambda}\,\sum_{k=0}^{\infty}\,\frac{\lambda^{2k}}{(2K)!}=e^{-\lambda}\,\left(\frac{e^{-\lambda}+e^{\lambda}}{2}\right)=\frac{1+e^{-2\,\lambda}}{2}
\end{equation*}
\end{proof}\,\\
\begin{tcolorbox}[ 
	title = \textcolor{black}{\textcolor{white}{Problema 8}},]
\textit{Si $X$ es una v.a. geom\'etrica, prueba que:
$\mathbb{P}[X=n+k|X>n]=\mathbb{P}[X=k].$
}
\end{tcolorbox}
\begin{proof}\,\\
  Primero aplicando la definici\'on de probabilidad total:\,\\
  \,\\
  \begin{equation*}
    P[X=n+k|X>n]=\frac{P[(X=n+k)\cap (X>n)]}{P[X>n]}
  \end{equation*}\,\\
  Comos sabemos la funci\'on de densidad de $X=k$ con $k$ entero mayor a uno es $P[X=k]=p(1-p)^{k-1}$ pa algun $p\in \R$, $0<p<1$, ahora como
  $k\geq 1$ entonces $X=n+k\geq n+1>n$, por tanto $((X=n+k)\cap (X>n))=X=n+k$, reemplazando en lo anterior:\,\\
  \,\\
  \begin{equation*}
    P[X=n+k|X>n]=\frac{P[X=n+k]}{P[X>n]}=\frac{p\,(1-p)^{n+k-1}}{P[X>n]}
  \end{equation*}\,\\
  Luego tenemos que:\,\\
  \begin{equation*}
    P[X>n]=\sum_{i=1}^{\infty}\,p\,(1-p)^{n+i-1}=p\,(1-p)^{n}\,\sum_{i=1}^{\infty}\,(1-p)^{i-1}=p\,(1-p)^{n}\,\sum_{l=0}^{\infty}\,(1-p)^{l}
  \end{equation*}\,\\
  Como $1-p<1$ tenemos una serie geometrica por tanto:\,\\
  \begin{equation*}
    P[X>n]=p\,(1-p)^{n}\,\sum_{l=0}^{\infty}\,\frac{1}{1-(1-p)}\,p\,(1-p)^n=(1-p)^n
  \end{equation*}\,\\
  Por tanto tenemos que:\,\\
  \begin{equation*}
     P[X=n+k|X>n]=\frac{p\,(1-p)^{n+k-1}}{(1-p)^n}=p\,(1-p)^{k-1}=P[X=k]
  \end{equation*}
\end{proof}
