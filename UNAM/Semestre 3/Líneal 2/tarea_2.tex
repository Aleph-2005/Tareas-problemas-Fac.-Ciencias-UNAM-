\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 1}},]
\textit{Para cada uno de los siguientes espacios con producto interior $V$ (Sobre $\mathbb{F}$) y transfromaciones lineales $g:V\rightarrow \mathbb{F}$, encontrar 
un vector $y$ tal que:\,\\
\,\\
\begin{equation*}
    g(x)=\langle x,y \rangle\,\,\,\,\forall x\in V
\end{equation*}
\begin{itemize}
    \item $V=\R^3$,\,\,\,\,$g(a_1,a_2,a_3)=a_1-2a_2+4a_3$
    \item $V=\C^2$\,\,\,\,\,$g(z_1,z_2)=2z_1+iz_2$
\end{itemize}
}
\end{tcolorbox}

\begin{Solucion}
Para el primer inciso, proponemos $y=(1,-2,4)$, sea $x=(a_1,a_2,a_3)\in \R^3$, entonces:
\begin{align*}
    \langle x,y \rangle &= \langle (a_1,a_2,a_3), (1,-2,4) \rangle \\
    &= a_1 -2a_2 + 4a_3 \\
    &= g(x)
\end{align*}
Como $x$ fue arbitrario, se tiene que $g(x)=\langle x,y \rangle$ para todo $x\in V$.\\[10pt]
Para el segundo inciso, proponemos $y=(2,-i)$, sea $x=(z_1,z_2)\in \C^2$, entonces:
\begin{align*}
    \langle x,y \rangle &= \langle (z_1,z_2), (2,-i) \rangle \\
    &= z_1 \overline{2}+ z_2(\overline{-i})  \\
    &= 2z_1 + iz_2 \\
    &= g(x)
\end{align*}
Como $x$ fue arbitrario, se tiene que $g(x)=\langle x,y \rangle$ para todo $x\in V$.\\[10pt]
\end{Solucion}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}},]
\textit{Sea $T$ un operador lineal en un espacio con producto interior $V$. Sean $U_1T+T^*$ y $U_2=T-T^*$. Demostra que:
\begin{equation*}
    U_1^*=U_1\,\,\,\,\,U_2=U_2^*
\end{equation*}
}
\end{tcolorbox}

\begin{Solucion}
    Probaremos que dados $S,T \in V^*$, se cumple que $(S\pm T)^*=S^*\pm T^*$.\\[10pt]
    Sean $x,y\in V$, entonces:
    \begin{align*}
        \PrdI{(S\pm T)(x)}{y} &= \PrdI{S(x)\pm T(x)}{y}  \\
        &=\PrdI{S(x)}{y} \pm \PrdI{T(x)}{y} \\
        &=\PrdI{x}{S^*(y)} \pm \PrdI{x}{T^*(y)} \\
        &=\PrdI{x}{S^*(y)\pm T^*(y)} \\
        &=\PrdI{x}{(S^*\pm T^*)(y)}
    \end{align*}
    Por lo tanto $(S\pm T)^*=S^*\pm T^*$.\\[10pt]
    Se sigue que:
\begin{multicols}{2}
    \begin{align*}
        U_1^* &= (T+T^*)^* \\
        &= T^* + (T^*)^* \\
        &= T^* + T \\
        &=T+T^* \\
        &= U_1
    \end{align*}

    \begin{align*}
        U_2^* &= (T-T^*)^* \\
        &= T^* - (T^*)^* \\
        &= T^* - T \\
        &=-(T-T^*) \\
        &= U_2
    \end{align*}
\end{multicols}

\end{Solucion}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 3}},]
\textit{Dar un ejemplo de un operador lineal $T$ en un espacio con producto interior tal que:
\begin{equation*}
    N(T)\neq N(T^*)
\end{equation*}
}
\end{tcolorbox}

\begin{Solucion}
    Sea $S,T: \R^2 \rightarrow \R^2$ definido por $T(x,y)=(x+y,-x-y)$ y $S(x,y)=(x-y,x-y)$. Veamos que ambos son lineales: \\
    Sean $(x_1,y_1),(x_2,y_2)\in \R^2$ y $c\in \R$, entonces: 
\begin{align*}
    T((x_1,y_1)+c (x_2,y_2)) &= T(x_1+cx_2,y_1+cy_2) \\
    &=(x_1+cx_2+y_1+cy_2,-(x_1+cx_2+y_1+cy_2)) \\
    &=(x_1+y_1,-(x_1+y_1))+c(x_2+y_2,-(x_2+y_2)) \\
    &=T(x_1,y_1)+cT(x_2,y_2)
\end{align*}

\begin{align*}
    S((x_1,y_1)+c (x_2,y_2)) &= S(x_1+cx_2,y_1+cy_2) \\
    &=(x_1+cx_2-(y_1+cy_2),x_1+cx_2-(y_1+cy_2)) \\
    &=(x_1-y_1,x_1-y_1)+c(x_2-y_2,x_2-y_2) \\
    &=S(x_1,y_1)+cS(x_2,y_2)
\end{align*}
    Aseguramos que $T^*=S$. Sean $(x,y),(z,w)\in \R^2$, entonces:
    \begin{align*}
        \PrdI{T(x,y)}{(z,w)} &= \PrdI{(x+y,-x-y)}{(z,w)} \\
        &= (x+y)z + (-x-y)w \\
        &= x(z-w) + y(z-w) \\
        &= \PrdI{(x,y)}{(z-w,z-w)} \\
        &= \PrdI{(x,y)}{S(z,w)}
    \end{align*}
    Ahora tenemos que:
    \begin{align*}
        T(1,-1) &= (0,0) \\
        S(1,-1) &= (2,2)
    \end{align*}
    Es decir $(1,-1)\in N(T)$ pero $(1,-1)\notin N(S)$, por lo tanto $N(T)\neq N(T^*)$.
\end{Solucion}\,\\

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 4}},]
\textit{Sea $T$ un operador lineal en un espacio con producto interior $V$. Demostrar que $T$ es invertible si y solo si $T^*$ es invertible y:
\begin{equation*}
    (T^*)^{-1}=(T^{-1})^*
\end{equation*}}
\end{tcolorbox}
\begin{proof}
    Supongamos que $T$ es invertible, sabemos que $T^{-1}$ es un operador l\'ineal por tanto $(T^{-1})^{*}$ existe, a su vez $T^{*}$ tambi\'en existe, usando las propiedades
    de ambos adjuntos se sigue que:\,\\
    \,\\
    \begin{equation*}
        \langle x,y\rangle=\langle T^{-1}(T(x)),y \rangle=\langle T(x), (T^{-1})^*(y) \rangle=\langle x, T^*((T^{-1})^*(y)) \rangle
    \end{equation*}\,\\
    Usando propiedades del producto interno:\,\\
    \,\\
    \begin{align*}
       0=\langle x,y\rangle-\langle x,T^*((T^{-1})^*(y)) \rangle=\langle x,y \rangle +\overline{(-1)}\langle x,T^*((T^{-1})^*(y)) \rangle&=\langle x,y \rangle+\langle x,-T^*((T^{-1})^*(y)) \rangle\,\\
       \,\\
       =\langle x,y-T^*((T^{-1})^*(y))\rangle
    \end{align*}\,\\
    Como lo anterior es valido $\forall\,x,y\in V$, en part\'icular tenemos que:\,\\
    \,\\
    \begin{equation*}
       \langle y-T^*((T^{-1})^*(y)),y-T^*((T^{-1})^*(y))\rangle=0\iff y-T^*((T^{-1})^*(y))=0_{v}\iff T^*((T^{-1})^*(y))=y 
    \end{equation*}\,\\
    Por tanto tenemos que $(T^{-1})^*$ e sun inverso derecho de $T^{*}$, como $V$ es de dimensi\'on f\'inita tenemos que $(T^{-1})^*$ tambi\'en es inverso izquierdo, se sigue
    que $T^*$ es invertible y $(T^{*})^{-1}=(T^{-1})^*$\,\\
    \newpage
    Supongamos que $T^{*}$ es invertible por tanto $(T^{*})^{-1}$ es operador l\'ineal, de esto se tiene que $((T^*)^{-1})^{*}$ existe usando 
    las propiedades de ambos adjuntos\,\\
    \,\\
    \begin{equation*}
        \langle x,y\rangle=\langle T^{*}((T^*)^{-1}(x)),y \rangle=\langle (T^{*})^{-1}(x), (T^*)^*(y) \rangle
    \end{equation*}\,\\
    Como se demostro en clase si $T$ es operador lineal se cumple que $(T^{*})^{*}=T$, por tanto:\,\\
    \,\\
    \begin{equation*}
        \langle x,y\rangle=\langle T^{*}((T^*)^{-1}(x)),y \rangle=\langle (T^{*})^{-1}(x), T(y) \rangle=\langle x,((T^{*})^{-1})^*(T(y)) \rangle
    \end{equation*}\,\\
    Procediendo an\'alogamente como el caso anterior tenemos que:\,\\
    \,\\
    \begin{equation*}
        \langle ((T^{*})^{-1})^*(T(y))-y,((T^{*})^{-1})^*(T(y))-y \rangle=0\iff ((T^{*})^{-1})^*(T(y))-y=0_V\iff ((T^{*})^{-1})^*(T(y))=y
    \end{equation*}\,'\\
    Por tanto $((T^*)^{-1})^*$ es inverso izquierdo de $T$ como $V$ es de dimensi\'on f\'inita se sigue que $T$ es invertible y $T^{-1}=((T^*)^{-1})^*$, como $T^{-1}$ es operador l\'ineal $(T^{-1})^*=(((T^*)^{-1})^*)^*$ existe, a su vez como $((T^*)^{-1})^*$ es operador l\'ineal se cumple que
    $(T^{-1})^*=(((T^*)^{-1})^*)^*=(T^*)^{-1}$
\end{proof}\,\\

    N(T^*\circ T)=N(T)

    \,\\
    \begin{equation*}
        \forall\,x,y\in V\,\,(\langle T(x),y \rangle=\langle x, T^{*}(y) \rangle)
    \end{equation*}\,\\
    En particular si elegimos $y=T(x)$ se obtiene que :\,\\
    \,\\
    \begin{equation*}
        \langle T(x),T(x) \rangle=\langle x, T^{*}(T(x)) \rangle=\langle x,0_{V} \rangle=
    \end{equation*}\,\\
    Como $\langle . \rangle$ es producto interno se sigue que:\,\\
    \,\\
    \begin{equation*}
        T(x)=0_V \iff  \langle T(x),T(x) \rangle=0
    \end{equation*}\,\\

	title = \textcolor{black}{\textcolor{white}{Problema 6}},]

    \item Demostrar que $T$ es lineal
    \item Demostrar que $T^*$ existe y dar su expresi\'on
\end{itemize}}
\end{tcolorbox}
\begin{proof}
   \textbf{i)}\,\,Sabemos que $\langle \lrcorner ,x\rangle:V\rightarrow F$ es lineal $\forall\,\,x\in V$, luego definimos, para $y\in V$ $T_{y}:F\rightarrow V$, de tal forma que
    $\alpha \Rightarrow \alpha y$, la cual es lineal por las propiedades del producto por escalar en $V$ por tanto $T_y\circ \langle \lrcorner ,x \rangle:V\rightarrow V$ es lineal $\forall x,y\in V$, pues es 
    composici\'on de funciones lineales finalmente notemos que $T(z)=\langle z,x \rangle y=T_y(\langle z,x \rangle)=T_y\circ \langle \lrcorner ,x \rangle(z)\,\,\forall x,y,z\in V $, por tanto $T$ es un operador lineal \,\\
    \,\\
    \textbf{ii)}\,\,Como $T$ es operador l\'ineal sabemos que existe $T^*$, finalmente como $T^*$ es el unico operador l\'ineal que cumple que:\,\\
    \,\\
    \begin{equation*}
        \forall\,x,y\in V\,\,(\langle T(x),y \rangle=\langle x, T^*(y) \rangle)
    \end{equation*}\,\\
    Basta encontrar un operador l\'ineal que cumpla esta propiedad proponemos $T^{*}(w)=\langle w,y \rangle x$, sabemos que $T^{*}$ es lineal por la misma raz\'on que $T$ lo es y ahora comprobemos que:\,\\
    \,\\
    \begin{equation*}
        \forall\,\,z,w\in W\,\,(\langle z,T^*(w)\rangle=\langle z, \langle w,y \rangle\,x \rangle=\overline{\langle w,y \rangle}\,\langle z,x \rangle=\langle z,x \rangle \langle y,w \rangle=\langle\langle z,x \rangle\,y,w \rangle=\langle T(z),w\rangle)
    \end{equation*}
\end{proof}\,\\
