\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 1}},]
\textit{Sea :
\begin{equation*}
    X=\{f:\R\to \R|f(x)=ax+b,a\in R\setminus \{0\},b\in \R\}
\end{equation*}
Demuestre que $X$, con la composici\'on usual de funciones, es un grupo.
}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}},]
\textit{Sea $(G,*,e)$ un grupo. Demuestra que el \'unico elemento $g\in G$ tal que $g^2=g$ es $e$
}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 3}},]
\textit{Sea $(G,*,e)$. Demuestra que si $g^2=e$ para todo $g\in G$ entonces $(G,*,e)$ es abeliano}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 4}},]
\textit{Sea $G$ un conjunto finito con una operaci\'on binaria $*$, con neutro $e$. Demuestre que $G$ es grupo si y s\'olo si la tabla 
de $*$ tiene las siguientes propiedades:\,\\
\begin{enumerate}
    \item Todo rengl\'on y cada columna de la tabla de $*$ contiene todos los elementos de $G$
    \item Para cualquier $x,y\in G\setminus \{e\}$ y para cada rect\'angulo $R$ en el cuerpo de la tabla que tenga a $e$ en uno de los v\'ertices, a 
    $x$ en el v\'ertice del mismo rengl\'on que $e$ y a $y$ en el v\'ertice de la misma columna que $e$, el cuarto v \'ertice est\'a determinado unicamente por el par $(x,y)$
\end{enumerate}
}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 5}},]
\textit{Sea
\begin{equation*}
    G=\{f:\R\to \R|f\text{es biyectiva}\}
\end{equation*}
con la operaci\'on de composici\'on. Considere
\begin{equation*}
    H=\{g\in G|f(x)=x+b,b\in \R\}
\end{equation*}
Demuestre que $H$ es un subgrupo de $G$}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 6}},]
\textit{Sea $G=(\R^{\N},+)$, el conjunto de todas las sucesiones reales con la suma coordenada a coordenada. Defina
\begin{equation*}
    H=\{(a_n)_{n\in \N}|\exists N\in \N \text{tal que} a_n=0 \text{para todo} n\geq N \}
\end{equation*}
Demuestre que $H$ es subgrupo de $G$}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 7}},]
\textit{Sea $G$ un grupo y $H\subset G$ un subconjunto no vac\'io. Demuestre que $H$ es un subgrupo de $G$ si y solo si para todo $a,b\in H$ se cumple que:
\begin{equation*}
    ab^{-1}\in H
\end{equation*}}
\end{tcolorbox} 

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 8}},]
\textit{Sea $G$ un grupo y sea $\{H_i\}_{i\in I}$ una familia de subgrupos de $G$ tal que:
\begin{equation*}
    \forall\,i,j\in I\,\,\:\exists\,k\in I con H_i\cup H_j\subset H_k
\end{equation*}
Demuestre que\,\\
\begin{equation*}
    H=\bigcap_{i\in I}\,H_i
\end{equation*}
es un subgrupo de $G$}
\end{tcolorbox} 

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 9}},]
\textit{Demuestra que el rec\'iproco del ejercicio anterior es falso. Es decir, construya un grupo
$G$ y una familia de subgrupos $\{H_i\}_{i\in I}$ tales que:\,\\
\begin{equation*}
    H=\bigcap_{i\in I}H_i
\end{equation*}
sea un subgrupo de $G$, pero que no se cumpla que para todos $i,j\in I$ exista $k\in I$ con:
\begin{equation*}
    H_i\cup H_j\subset H_k
\end{equation*}
justifique cuidadosamente que su ejemplo satisface ambas propiedades
}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 10}},]
\textit{Sea $G$ un grupo y sea $H$. defina:
\begin{equation*}
    C_G(H)=\{g\in G|gh=gh \text{para todo} h\in H\}
\end{equation*}
Demuestra que $C_G(H)$ es un subgrupo de $G$
}
\end{tcolorbox}
