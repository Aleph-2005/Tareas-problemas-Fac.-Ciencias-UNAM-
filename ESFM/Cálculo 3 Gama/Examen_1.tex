\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}}]
\textit{Sea $\chi$ un conjunto no vacio, decimos que una colecci\'on $\zeta\subset 2^{\chi} $ cumple la propiedad
de la intersecci\'on finita cuando para cada $A\subset \zeta$ de cardinalidad finita se cumple que:\,\\
\,\\
\begin{equation*}
    \bigcap_{B\in A}\,B \neq \emptyset
\end{equation*}\,\\
Demostrar que si $A\subset \R^n$ es compacto , entonces para cada $\chi\subset 2^A$ que cumpla la propiedad de la intersecci\'on finita
y que para cada $B\in \chi$ existe $U_{B}\in \R^n$ conjunto cerrado tal que $B=U_{B}\cap A$, se cumple que:\,\\
\,\\
\begin{equation*}
    \bigcap_{C\in\chi}\,C\neq \emptyset
\end{equation*}
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
Procedemos por contradicci\'on, es decir existe $\chi\subset 2^A$ que cumple la propiedad de la intersecci\'on finita y que para todo $B\in \chi$ existe $U_B\in \R^n$ cerrado tal que $B=U_B\cap A$ que ademas cumple con:\,\\
\,\\
\begin{equation*}
    \bigcap_{B\in\chi}\,B=\emptyset
\end{equation*}\,\\
Si sacamos el complemento respecto a $A$:\,\\
\,\\
\begin{equation*}
    A=\emptyset^c=\left(\bigcap_{B\in\chi}\,B\right)^c=\bigcup_{B\in\chi}\,B^c=\bigcup_{B\in \chi}\,(U_B\cap A)^c
\end{equation*}\,\\
Como $U_B$ es cerrado en $\R^n$ y $A$ es cerrado en $\R^n$ se tiene que $U_B\cap A$ es cerrado en $\R^n$ y por tanto $(U_B\cap A)^c$ es abierto en $\R^n$ para todo $B\in \chi$, por tanto $A=\bigcup_{B\in \chi}\,(U_B\cap A)^c$ es una cubierta abierta de $A$, como 
$A$ es compacto existe $\zeta \subset \chi$ de cardinalidad finita tal que:\,\\
\,\\
\begin{equation*}
    A=\bigcup_{B\in \zeta}\,(U_B\cap A)^c
\end{equation*}\,\\
Sacando de nuevo el complemento respecto a $A$:\,\\
\,\\
\begin{equation*}
    \emptyset=A^c=\left(\bigcup_{B\in \zeta}\,(U_B\cap A)^c\right)^c=\bigcap_{B\in\zeta}\,(U_B\cap A)=\bigcap_{B\in \zeta}\,B
\end{equation*}\,\\
Pero como $\chi$ cumple la propiedad de la intersecci\'on finita y $\zeta\subset \chi$ de cardinalidad finita entonces:\,\\
\,\\
\begin{equation*}
    \emptyset\neq\bigcap_{B\in \zeta}\,B=\emptyset
\end{equation*}\,\\
Una contradicci\'on, como nuestra unica suposici\'on fue la existencia de $\chi$ concluimos la tesis del problema.
\end{proof}
