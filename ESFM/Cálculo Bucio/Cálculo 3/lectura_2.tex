\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 1}}]
\textit{Sean $p,q,r>1$ en $\R$ tal que $\frac{1}{p}+\frac{1}{q}=\frac{1}{r}$ y sean $x,y\in \R^n$.\textbf{Demuestre} que\,\\
\,\\
\begin{equation*}
    \left(\sum_{i=1}^n\,|x_iy_i|^r\right)^{\frac{1}{r}}\leq \parallel x \parallel_{p}\,\,\parallel y \parallel_{q}
\end{equation*}
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
    Consideremos el vector $|x|^r=(|x_1|^r,|x_2|^r,\cdots,|x_n|^r)$  para cualquier $x\in \R^n$, luego notemos que $\frac{r}{p}+\frac{r}{q}=1$, ademas que $\frac{p}{r},\frac{q}{r}>1$, luego aplicando la desigualdad de Holder a $x,y\in \R^n$:\,\\
    \,\\
    \begin{equation*}
        \sum_{i=1}^n\,|x_iy_i|^r=\langle\,|x|^r,|y|^r\rangle\,\,\leq\,\,\| |x|^r \|_{\frac{p}{r}}\,\,\| |y|^r \|_{\frac{q}{r}}=\left(\sum_{i=1}^n (|x_i|^r)^{\frac{p}{r}}\right)^{\frac{r}{p}}\,\left(\sum_{i=1}^{n}\,(|y_i|^r)^{\frac{q}{r}}\right)^{\frac{r}{q}}=(\|x\|_{p}\,\,\|y\|_{q})^r
    \end{equation*}\,\\
    \,\\
    Finalmente sacando  raiz $r$-esima tenemos que:\,\\
    \,\\
    \begin{equation*}
        \left(\sum_{i=1}^n\,|x_iy_i|^r\right)^{\frac{1}{r}}\,\,\leq\,\,\|x\|_{p}\,\,\|y\|_{q}
    \end{equation*} 
    
\end{proof}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}}]
\textit{Sean $p,q> 1$ en $\R$ tales que $p<q$.\,\,\textbf{Demuestre} que para todo $x\in \R^n$ se tiene que:\,\\
\,\\
\begin{equation*}
    \|x\|_q\,\,\leq\,\,\|x\|_p\,\,\leq n^{\frac{p}{q}}\,\,\|x\|_q
\end{equation*}
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
    Aplicando el problema anterior con $s=\frac{1}{p}-\frac{1}{q}>0$ y $\frac{1}{q}$ $\left(s+\frac{1}{q}=\frac{1}{p}\right)$ tenemos que para cualesquiera $x,y\in \R^n$\,\\
    \,\\
    \begin{equation*}
        \left(\sum_{i=1}^n|x_iy_i|^p\right)^{\frac{1}{p}}\leq \|x\|_{s^{-1}}\,\|y\|_q
    \end{equation*}\,\\
    \,\\
    Luego si tomamos $x=\bar{1}=(1,1,\cdots,1)\in \R^n$, tenemos que:\,\\
    \,\\
    \begin{equation*}
        \|y\|_p= \left(\sum_{i=1}^n|y_i|^p\right)^{\frac{1}{p}}\leq n^{s}\,\|y\|_q
    \end{equation*}\,\\
    Adicionalmente tenemos que $\frac{q}{p}>s>0$ y por la monotonia de la exponencial se concluye que:\,\\
    \,\\
    \begin{equation*}
        n^s<n<n^{\frac{p}{q}}
    \end{equation*}\,\\
    Por tanto:\,\\
    \begin{equation*}
        \forall x\in \R^n (\|x\|_p\leq n^{\frac{q}{p}}\|x\|_q)
    \end{equation*}
\end{proof}
