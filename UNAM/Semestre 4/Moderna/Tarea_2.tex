\begin{tcolorbox}[

\textit{Sea $G$ un grupo y sea $\{H_i\}_{i\in I}$ una familia de subgrupos de $G$ tal que:
\begin{equation*}
    \forall\,i,j\in I\,\,\:H_i\subseteq H_j\,\,\text{o}\,\,H_j\subset H_i
\end{equation*}
Demuestre que\,\\
\begin{equation*}
    H=\bigcup_{i\in I}\,H_i
\end{equation*}
es un subgrupo normal de $G$}
\end{tcolorbox}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 4}},]
\textit{Sea $H$ el unico subgrupo finito de $G$ de orden $n$. Demuestra que si $H$ es el unico subgrupo
de $G$ de indice $n$, entonces $H$ es subgrupo normal de $G$
}
\end{tcolorbox}
\begin{proof}
   Sea $g\in G$ definimos $C_g:G\to G$ con $C_g(h)=g^{-1}hg$, en clase se demostro que este es un automorfismo de $G$, ahora es f\'acil notar que
   $C_g[H]=g^{-1}Hg$, por tanto si $H$ es subgrupo de $G$ entonces $C_g[H]=g^{-1}Hg$ tambien es subgrupo de $G$, pues $C_g$ es epimorfismo, luego $|C_g[H]|=|H|=n$ pues $C_g$ es biyectiva, se tiene que necesariamente $g^{-1}Hg=H$, como $g\in G$ fue arbitraria
   concluimos que $H\trianglelefteq G$ 
\end{proof}


\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 5}},]
\textit{Sea $H\leq K\leq G$. Demuestra que si $K$ es un subgrupo ciclico normal de $G$, entonces $H$ es subgrupo normal de $G$}
\end{tcolorbox}
\begin{proof}
    \textbf{Primera Soluci\'on}\,\\
    Es claro que dado $g\in G$ $g^{-1}Hg\subset g^{-1}k =K$, pues $K\trianglelefteq G$, luego por el problema anterior tenemos que 
    $g^{-1}Hg\leq K$, ademas $|g^{-1}Hg|=|H|$, como $K$ es un subgrupo ciclico y finito, este tiene un \'unico subgrupo de orden $|H|$, de tal forma que
    $g^{-1}Hg=H$, de nuevo como $g\in G$ fue arbitraria se sigue que $H\trianglelefteq G$  
\end{proof}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 7}},]
\textit{Demuestra que el generado de la uni\'on de subgrupos normales es un subgrupo normal }
\end{tcolorbox}
\begin{proof}
    Sea $\{H_i\}_{i\in I}$ una colecci\'on de subgrupos normales de $G$, sabemos que si $x\in\left\langle \displaystyle\bigcup_{i\in I}\,H_I\right\rangle$, entonces $x$ se puede escribir como producto finito de elementos $(x=a_1a_2\cdots a_n)$ con $a_k\in \displaystyle\bigcup_{i\in I}H_i$ para toda $k\in \N_n$, por tanto tenemos que para $g\in G$
    \begin{equation*}
        g^{-1}xg=g^{-1}a_1a_2\cdots a_ng=g^{-1}a_1gg^{-1}a_2g^{-1}g\cdots gg^{-1}a_ng
    \end{equation*}
    Como cada $H_i$ es normal se sigue que $g^{-1}a_kg\in \displaystyle\bigcup_{i\in I}H_i$ para todo $k\in \N_n$, por tanto $g^{-1}xg\in \displaystyle\bigcup_{i\in I}H_i$ y por tanto $g^{-1}xg\in\left\langle \displaystyle\bigcup_{i\in I}H_i\right\rangle$, por tanto $\left\langle\displaystyle\bigcup_{i\in I}\,H_I\right\rangle\trianglelefteq G$
\end{proof}
