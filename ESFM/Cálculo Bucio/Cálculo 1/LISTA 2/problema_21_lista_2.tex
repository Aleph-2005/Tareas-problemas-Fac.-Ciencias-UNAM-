\begin{tcolorbox}[title = \textcolor{black}{\textcolor{white}{Problema 21}}]
\textit{\textbf{Demuestra} por inducci\'on sobre $n\in \N$ que:\,\\
\begin{equation*}
    \sqrt{(m+1)!+\sqrt{(m+2)!+\sqrt{(m+3)!+\cdots+\sqrt{(m+n)!}}}}\leq (m+1)!+1
\end{equation*}\,\\
para todo $m,n\in \N$}
\end{tcolorbox}
\begin{proof}
    Procedemos por inducci\'on sobre $n\in \N$:
    \begin{itemize}
        \item \textbf{Base de inducci\'on}: Es claro que:\,\\
       \begin{equation*}
        \sqrt{(m+1)!}\leq (m+1)!\leq (m+1)!+1\,\,\forall\,m\in \N
       \end{equation*}\,\\
       \item \textbf{Hip\'otesis de inducci\'on}: Supongamos que existe $k\in \N$ tal que para todo $m\in \N$ se cumple que:\,\\
       \begin{equation*}
        \sqrt{(m+1)!+\sqrt{(m+2)!+\sqrt{(m+3)!+\cdots+\sqrt{(m+k)!}}}}\leq (m+1)!+1
       \end{equation*}\,\\
       \item \textbf{Paso inductivo}:\,\\
        \begin{equation*}
            \sqrt{(m+1)!+\underbrace{\sqrt{(m+2)!+\sqrt{(m+3)!+\cdots+\sqrt{(m+k)!+\sqrt{(m+k+1)!}}}}}_\text{k veces}}\,\,\overset{\textbf{H.I}}{\leq} \sqrt{(m+1)!+(m+2)!+1}
        \end{equation*}\,\\
        Ahora veamos que:\,\\
        \begin{align*}
            (m+2)\leq (m+1)!+1&\implies (m+1)!(m+2)\leq (m+1)![(m+1)!+1]\\
            \,\\
            \implies (m+2)!\leq (m+1)!^2+(m+1)!&\implies (m+2)!+(m+1)!+1\leq (m+1)!^2+2(m+1)!+1\\
            \,\\
            \implies \sqrt{(m+1)!+(m+2)!+1}&\leq (m+1)!+1
        \end{align*}\,\\
        Por tanto:\,\\
        \begin{equation*}
            \sqrt{(m+1)!+\sqrt{(m+2)!+\sqrt{(m+3)!+\cdots+\sqrt{(m+k)!+\sqrt{(m+k+1)!}}}}}\leq (m+1)!+1
        \end{equation*}
    \end{itemize}
\end{proof}\,\\
\begin{tcolorbox}[title = \textcolor{black}{\textcolor{white}{Problema 21}}]
\textit{\textbf{Demuestra} que la sucesi\'on $\{a_n\}_{n\in \N}$ donde:\,\\
\begin{equation*}
\sqrt{1!+\sqrt{2!+\sqrt{3!+\cdots+\sqrt{n!}}}}\,\,\,\,\,\forall\,\,n\in \N
\end{equation*}\,\\
es convergente}
\end{tcolorbox}
\begin{proof}
    Por el problema anterior tenemos que para toda $n\in \N$:\,\\
    \begin{equation*}
        \sqrt{1!+\sqrt{2!+\sqrt{3!+\cdots+\sqrt{n!}}}}\leq (0+1)!+1=2
    \end{equation*}\,\\
    Luego tenemos que para $n\in\N$:\,\\
    \begin{align*}
        \sqrt{n!}&\leq \sqrt{n!+\sqrt{(n+1)!}}\,\\
        \,\\
        \implies (n-1)!+\sqrt{n!}&\leq (n-1)!+\sqrt{n!+\sqrt{(n+1)!}}\,\\
        \,\\
        \implies \sqrt{(n-1)!+\sqrt{n!}}&\leq \sqrt{(n-1)!+\sqrt{n!+\sqrt{(n+1)!}}}\,\\
        \,\\
        &\cdots\text{$n$ veces}\,\\
        \,\\
        a_n=\sqrt{1!+\sqrt{2!+\sqrt{3!+\cdots+\sqrt{n!}}}}&\leq \sqrt{1!+\sqrt{2!+\sqrt{3!+\cdots+\sqrt{n!+\sqrt{(n+1)!}}}}}=a_{n+1}
    \end{align*}\,\\
    Es decir $\{a_n\}_{n\in \N}$ es mon\'otona y acotada luego convergente
\end{proof}
