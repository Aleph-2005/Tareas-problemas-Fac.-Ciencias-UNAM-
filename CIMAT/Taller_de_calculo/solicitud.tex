\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 1}},]
\textit{Define $f:(0,\infty)\to \R$ como $f(x)=2^{-x}+2^{-\frac{1}{x}}$
\begin{enumerate}
    \item C\'alcula $\displaystyle\lim_{x \to 0^{+}}f(x)$ y $\displaystyle\lim_{x\to \infty} f(x)$
    \item C\'alcula (y demuestra que tu c\'alculo es correcto) el m\'aximo de $f$ en todo su dominio
    \item ¿Qu\'e puedes decir  de su m\'inimo (tambi\'en global)?
\end{enumerate}}
\end{tcolorbox}
\begin{proof}
    \begin{enumerate}
        \item Vamos a demostrar lo siguiente:
        \begin{equation*}
            \lim_{x\to 0^{+}}2^{-\frac{1}{x}}=0
        \end{equation*}
        Primero notemos que la funci\'on $g(x)=2^x$ es no acotada y creciente, dado $\epsilon>0$ existe $n_1\in \N$ tal que $\frac{1}{n_1}<\epsilon$, luego existe $m\in \R^{+}$ tal que si $x\geq m$ entonces $2^x\geq n_1$, pues $g$ es creciente y no acotada, por tanto si $0<x<\frac{1}{m}$ entonces $\frac{1}{x}>m$ por tanto:\,\\
        \begin{equation*}
            2^{\frac{1}{x}}\geq n_1\implies 2^{-\frac{1}{x}}\leq \frac{1}{n_1}<\epsilon
        \end{equation*}
        Por tanto concluimos que $\displaystyle \lim_{x\to 0^{+}}2^{-\frac{1}{x}}=0$
        ahora la funci\'on $2^{-x}$ es continua en todo $\R$ por tanto $2^{-0}=1=\displaystyle \lim_{x\to 0}2^{-x}=\displaystyle \lim_{x\to 0^{+}}2^{-x}$, por tanto:\,\\
        \begin{equation*}
            \displaystyle \lim_{x\to0^{+}}2^{-x}+2^{-\frac{1}{x}}=\displaystyle \lim_{x\to 0^{+}}2^{-x}+\displaystyle \lim_{x\to0^{+}}2^{-\frac{1}{x}}=1+0=1
        \end{equation*}
        Ahora demostraremos que $\displaystyle \lim_{x\to \infty}2^{-x}=0$, primero dado $\epsilon>0$ existe $n\in\N$ tal que $\frac{1}{n}<\epsilon$, luego como $2^x$ es una funci\'on no creciente y acotada existe $m\in \R^{+}$ tal que si $x\geq m$ entonces $2^x\geq n\implies \frac{1}{2^x}\leq \frac{1}{n}<\epsilon$ por tanto $\displaystyle \lim_{x\to \infty}2^{-x}=0$
        luego notemos que como $\displaystyle\lim_{x\to \infty}\frac{1}{x}=0$, en $\displaystyle\lim_{x \to \infty}2^{-\frac{1}{x}}$ podemos hacer el cambio de variable $u=\frac{1}{x}$, luego si $x\to \infty$ entonces $u\to 0$ por tanto $\displaystyle\lim_{x \to \infty}2^{-\frac{1}{x}}=\displaystyle \lim_{u\to 0}2^{-u}$, usando la continuidad de $2^{-x}$ en $x=0$ concluimos que $\displaystyle\lim_{x \to \infty}2^{-\frac{1}{x}}=\displaystyle \lim_{u\to 0}2^{-u}=2^{-0}=1$, 
        por tanto:\,\\
        \begin{equation*}
            \displaystyle\lim_{x\to \infty}2^{-x}+2^{-\frac{1}{x}}=\displaystyle \lim_{x\to \infty}2^{-x}+\displaystyle \lim_{x\to\infty}2^{-\frac{1}{x}}=0+1=1
        \end{equation*}
        \newpage
        \item Para encontrar los m\'aximos de la funci\'on procedemos de la manera mas natural encontrando los puntos criticos, para ello como $f$ es derivable en su dominio, solo tenemos que analizar las raices de su derivada para encontrar los puntos criticos, pues como es derivable en dominio abierto cualquier maximo o minimo sera punto critico:
        \begin{equation*}
            f'(x)=ln(2)\left(\frac{2^{-\frac{1}{x}}}{x^2}-2^{-x}\right)
        \end{equation*}
        Notemos que una condici\'on necesaria y suficiente para $f'(x)=0$ es que:\,\\
        \begin{equation*}
            \frac{2^{-\frac{1}{x}}}{x^2}-2^{-x}=0\implies 2^{x-\frac{1}{x}}=x^2\implies ln(2)\left(x-\frac{1}{x}\right)-2ln(x)=0
        \end{equation*}
        Por tanto encontrar puntos critcios de $f$ es equivalente a encontrar raices de la funci\'on\\$h(x)=ln(2)\left(x-\frac{1}{x}\right)-2ln(x)$
        ahora antes de continuar con el an\'alisis haremos uso de un pequeño lema:\,\\
        \begin{lemma}
            Sea $h$ una funci\'on con derivada continua $h'$ si $h'$ tiene solo $2$ raices $f$ entonces tendra a lo m\'as $2+1=3$ raices
        \end{lemma}
        \begin{proof}
            Supongamos por el contrario que $h$ tiene $4$ raices $x_1,x_2,x_3,x_4$ como, tenemos que $h$ es derivable en $(x_1,x_2)$ y continua en $[x_1,x_2]$ ademas $h(x_1)=h(x_2)=0$ por el teorema de Rolle existe $\bar{x}_1\in(x_1,x_2)$ tal que $h'(\bar{x}_1)=0$, notemos que podemos construir a $\bar{x}_2\in(x_2,x_3),\bar{x}_3\in(x_3,x_4)$ tales que $h'(\bar{x}_1)=h'(\bar{x}_2)=h'(\bar{x}_3)=0$, esto implicaria que $h$ tiene al menos tres raices una contradicci\'on pues $h$ tiene solo dos,
            por tanto concluimos que $h$ tiene a lo m\'as tres raices
        \end{proof}
        Ahora notemos que:\,\\
        \begin{equation*}
            h'(x)=ln(2)\left(1+\frac{1}{x^2}\right)-\frac{2}{x}=\frac{x^2ln(2)-2x+ln(2)}{x^2}
        \end{equation*}
        Por tanto las raices de $h'(x)$ estan unicamente determinadas por $x^2ln(2)-2x+ln(2)$ calculemos el discriminanteb para ver el n\'umero de soluciones\,\\
        \begin{equation*}
            \Delta=4-4ln(2)^2\approx2.078>0
        \end{equation*}
        Por tanto $h'(x)$ tiene unicamente dos raices en el intervalo $(0,\infty)$ por tanto usando el lema $0.1$ tenemos que $h(x)$ tiene a lo m\'as $3$ raices i.e $f$ tiene a lo m\'as tres puntos criticos.
        Procedemos a encontrar los puntos cr\'iticos de $f$, es claro que en $x=1$ se tiene un punto critico pues:\,\\
        \begin{equation*}
            f'(1)=ln(2)\left(\frac{2^{-1}}{1}-2^{-1}\right)=ln(2)\left(\frac{1}{2}-\frac{1}{2}\right)=0
        \end{equation*}
        Para determinar la naturaleza del punto critico evaluamos $f'$ en un punto en el intevalo $x_1\in(0,1)$ y otro en el intervalo $x_2\in(1,\infty)$:\,\\
        \begin{equation*}
            f'\left(\frac{3}{4}\right)=ln(2)\left(\frac{2^{-\frac{4}{3}}}{\frac{9}{16}}-2^{-\frac{3}{4}}\right)\approx0.0768>0
        \end{equation*}
        \begin{equation*}
            f'\left(\frac{4}{3}\right)=ln(2)\left(\frac{2^{-\frac{3}{4}}}{\frac{16}{9}}-2^{-\frac{4}{3}}\right)\approx-0.04324<0
        \end{equation*}
        por tanto $x=1$ es un m\'aximo local, ahora afirmamos que $f$ tiene otros dos punto criticos pues:
         \begin{equation*}
            f'(5)=ln(2)\left(\frac{2^{-\frac{1}{5}}}{25}-2^{-5}\right)\approx0.00247593>0
        \end{equation*}
        \begin{equation*}
            f'\left(\frac{1}{8}\right)=ln(2)\left(\frac{2^{-\frac{8}{1}}}{\frac{1}{64}}-2^{-\frac{1}{8}}\right)\approx-0.46233<0
        \end{equation*}
        Como $f'$ es continua y $f\left(\frac{1}{8}\right)<0,f\left(\frac{3}{4}\right)>0$ por teorema del valor intermedio existe $y_1\in\left(\frac{1}{8},\frac{3}{4}\right)$ tal que $f'(y_1)=0$ de la misma manera como $f\left(\frac{4}{3}\right)<0,f(5)>0$ existe $y_2\in\left(\frac{4}{3},5\right)$ tal que $f'(y_2)=0$, por tanto $x=1,y_1,y_2$ son los \'unicos puntos criticos de $f$ ademas por el signo de la derivada podemos afirmar que tanto $y_1,y_2$ son puntos minimos locales,
        ahora demostraremos que $f(x)\leq 1$ para todo $x\in(0,1]$, para ello primero notemos que en el intervalo $(0,y_1]$ la funci\'on es estrictamente decreciente pues no puede habe otro punto crictio entre $0$ y $y_1$, por tanto en este intervalo la funci\'on estara claramente acotada por $1$ pues $\displaystyle\lim_{x \to 0^{+}}=1$, luego tenemos que si existiera un punto $z\in(y_2,1]$ tal que $f(z)>1$, claramente $y_1<z<1<y_2$, luego como $f(y_2)<1$ y $f(z)>1$ por teorema del valor medio
        existe $d\in(y_2,z)$ tal que $f(d)=1$, ahora $f$ cumple las hipotesis del teorema de Rolle en el intervalo $[d,1]$, por tanto existiria $y_2<d<y_3<1<y_2$ con $y_3$ punto crictico, una contradicci\'on pues $f$ tiene solo tres puntos criticos, por tanto $f(x)\leq 1$ para todo $x\in(0,1]$
        ahora para demostrar que $f(x)\leq 1$ para todo $x\in[1,\infty)$, notemos que $f(x)=f\left(\frac{1}{x}\right)$, por tanto si $x>1\implies \frac{1}{x}<1$, por tanto $f(x)=f\left(\frac{1}{x}\right)<1$, por tanto $x=1$ no es solo un m\'aximo local sino global.
        \item Para abordar el tema del minimo global, podemos seguir el siguiente razonamiento, sea $\epsilon=\frac{1}{2}$ como $\displaystyle\lim_{x\to \infty}f(x)=1$, entonces existe $m\in \R^{+}$ tal que si $x\geq m$ entonces:\,\\
        \begin{equation*}
            |f(x)-1|< \frac{1}{2}\implies1-\frac{1}{2}<f(x)<1+\frac{1}{2}\implies \frac{1}{2}< f(x)
        \end{equation*}
        luego podemos extender $f$ continuamente del dominio $(0,m]$ a $[0,m]$ haciendo $f(0)=1$ pues $\displaystyle \lim_{x\to0^+}f(x)=1$, por tanto $f$ alcanza ssu m\'aximo y m\'inimo en $x_1\in[0,m]$ al ser continua, luego es claro que $min\left\{\frac{1}{2},f(x_1)\right\}\leq f(x)$, es decir $f$ tiene un minimo global
    \end{enumerate}
    \end{proof}

	title = \textcolor{black}{\textcolor{white}{Problema 2 (Parte 1)}},]
\textit{Considera la parabola $y=ax^2$. Toma sobre su gr\'afica dos puntos  cuales quiera $A=(x_1,y_1)$ y $B=(x_2,y_2)$. Considera la cuerda $\overline{AB}$. Prueba que existe 
un \'unico punto $C$ sobre la gr\'afica de la par\'abola tal que la recta tangente a la p\'arabola en $C$ es paralela a la cuerda $\overline{AB}$. Calcula las coordenadas de $C$  }
\end{tcolorbox}
\begin{proof}
    Para encontrar la pendiente de la cuerda $\overline{AB} $ simplemente usamos la formula punto pendiente:\,\\
    \begin{equation*}
        m=\frac{y_2-y_1}{x_2-x_1}=\frac{a(x_2^2-x_1^2)}{(x_2-x_1)}=a(x_1+x_2)
    \end{equation*}
    Ahora tenemos que $f'(x)=2ax$ por tanto si la derivada es paralela a la cuerda $\overline{AB}$:\,\\
    \begin{equation*}
        2ax=a(x_1+x_2)\iff x=\frac{x_1+x_2}{2}
    \end{equation*} 
    notemos que $x$ esta en funci\'on de las abcsisas $x_1, x_2$, por tanto este punto es \'unico, las coordenada del punto $C$ donde la derivada es paralela a $\overline{AB} $ son $C=\left(\frac{x_1+x_2}{2},a\frac{(x_2+x_1)^2}{4}\right)$
\end{proof}

es claro que $S\leq A$, pues $S$ es el supremo de las sumar parciales dadas por los triangulos hasta el enesimo paso y $A$ es una cota superior de estas sumas parciales, ahora solo faltaria demostrar que siempre  se puede aproximar una suma parcial tanto como se quiera a $A$\,\\

Para ello vamos a demostrar que en cualquier parabola $y=ax^2$, el \'area formada por un triangulo que tiene por verctices los extremos de una cuerda $AB$ y el punto $C$ de la parabola donde su recta tangente es paralela a la cuerda cumple que $area(\Delta ACB )>\frac{A}{2}$, para ello construimos la paralea a $AB$ que pasa por $C$ la que llamamos $g$, a su vez trazamos $k$ la recta paralela al diametro que pasa por $C$ e intercepta a la cuerda $AB$ en $M$, formamos los dos paralelogramos $\square AMCD$ y $\square MBEC$, notemos que las cuerdas $AC$ y $CB$ son sus respectivas diagonales, por lo cual dividen al paralelogramo en dos tri\'angulos de la misma area, notemos que el paralelogramo $\square AMCD$ se divide en los tri\'angulos $\Delta ADC$ y $\Delta MAC$, luego notemos que el arco de parabola $AC$ necesariamente queda contenido dentro de  
$\square AMCD$, por ello podemos afirmar que el area acotada por la parabola el segmento $AM$ y el segmento $MC$ la cual llamaremos $A_1$ cumple que $A_1<area(\square AMCD)=2area(\Delta AMC)$, de manera analoga podemos acotar el area delimitada por el arco $CB$ los segmentos $MB$ y $CM$ a la que llamaremos $A_2$ de la siguiente manera $A_2<area(\square MBEC)=2area(\Delta BMD)$, finalemente notamos que $A=A_1+A_2<2(area(\Delta AMC)+area(\Delta BMD))=2area(\Delta ACB)$

\end{figure}\,\\
\,\\
Ahora vamos a ver que $A-S_n<\frac{A-S_{n-1}}{2}$, donde $S_n$ es la $n$-esima suma parcial de las areas de los triangulos inscritos, para ello notemos que $S_n=S_{n-1}+\displaystyle\sum_{i=1}^{2^{n-1}}T_i$, donde $T_i$ son los tri\'angulos inscritos en el paso $n$, es claro que estos son en efecto $2^{n-1}$, pues cada tri\'angulo inscrito nos debide el segmendo parabolico a calcular en dos, cada uno de estos triangulos inscritos sobre los segmentos parabolicos dados en el paso $n-1$ a los que llamamos $D_i$ con $i\in\{1,\cdots,2^{n-1}\}$, cumplira que $T_i>\frac{D_i}{2}$, ahora es claro que:\,\\
\begin{equation*}
    \displaystyle\sum_{i=1}^{2^{n-1}}T_i>\frac{1}{2}\displaystyle\sum_{i=1}^{2^{n-1}}D_i=\frac{A-S_{n-1}}{2}\implies A-S_{n-1}- \displaystyle\sum_{i=1}^{2^{n-1}}T_i=A-S_n<A-S_{n-1}-\frac{A-S_{n-1}}{2}=\frac{A-S_{n-1}}{2}
\end{equation*}
Por tanto si definimos la suces\'ion $(a_n)_{n\in\N}$ con $a_n=A-S_n$, esta cumple que $a_n<\frac{a_{n-1}}{2}$ y por inducci\'on $a_n<\frac{a_1}{2^n}$ para todo $n\in \N$, demostraremos que esta sucesi\'on tiene limite igual a cero, como $\displaystyle \lim_{n\to \infty} \frac{1}{2^n}=0$, dado $\frac{\epsilon}{a_1}>0$ existe $K\in \N$ tal que si $n>K$ se tiene que:\,\\
\begin{equation*}
    a_n<\frac{a_1}{2^n}<\frac{\epsilon\,a_1}{a_1}=\epsilon\,\,(n\geq k)
\end{equation*}
Por tanto $\displaystyle \lim_{n\to \infty} a_n=0$, por lo tanto $A$ cumple que:\,\\
\begin{enumerate}
    \item $S_n<A$\,\,para todo $n\in \N$
    \item para todo $\epsilon>0$ existe $k\in \N$ tal que $A-\epsilon<S_k$
\end{enumerate}
i.e $A=S$ donde $S$ es el supremo de las sumas parciales dada por el area de los tri\'angulos inscritos en el $n$-esimo paso, es decir la uni\'on infinita de todos estos triangulos en efecto subre el area y solo el area deseada\,\\
\,\\
Finalmente por lo visto anteriormente en el e-$n$simo paso añadimos $\frac{area (ACB\Delta) }{4^n}$, por tanto el area del segmento de parabola acotado por la cuerda $AB$ es:\,\\
\begin{equation*}
    A=\sum_{i=0}^{\infty}\frac{area(\Delta ACB)}{4^i}=area(\Delta ACB)\,\left(\frac{1}{1-\frac{1}{4}}\right)=\frac{4\,area(\Delta ACB)}{3}
\end{equation*}
    \end{enumerate}
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2 (Parte 3)}},]
\textit{Explica por qu\'e el teorema es cierto para cualquier parabola, y no solo unicamente para $y=ax^2$ }
\end{tcolorbox}
\begin{proof}
    Esto es debido a que las clases de equivalencia de la parabola bajo la relaci\'on funciones afines (isometrias) esta dada justo por las parabolas canonicas $y=ax^2$, en palabras mas simples, cualquier otra parabola puede llevarse a traves de isometrias a una parabola de la forma $y=ax^2$, donde el teorema es valido, ahora como las isometrias no alteran el \'area de las figuras planas, tenemos que al regresar a su posici\'on original
    el teorema seguira siendo valido
\end{proof}
