\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 1}},]
\textit{Sea $T_F:V\rightarrow V$ l\'ineal y suponga que $\|T(x)\|=\|x\|$, para toda $x\in V$. Demuestre que $T$ es uno a uno
}
\end{tcolorbox}
\begin{proof}\,\\
	\,\\
    Sea $T:_FV\rightarrow_FV$ lineal, tal que $\|T(x)\|=\|x\|\quad \forall x \in _FV$.\\
Sean $\{ x,y\}\subset _FV$, tales que $T(x)=T(y)$, entonces por ser $T$ lineal, se tiene que:\\

\begin{align*}
    &T(x-y)=T(x)-T(y)=0\\
    &\Rightarrow \|T(x-y)\|=0\\
    &\Rightarrow \|x-y\|=0 \\
    &\Rightarrow x=y
\end{align*}
Es decir $T$ es inyectiva.\\[10pt]
Si $dim_F(V)=n,\; n\in \N$, entonces por ser inyectiva $T$ es biyectiva.
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}},]
\textit{Sea $W$ un espacio con producto interior y sea $T_F:V\rightarrow W$ l\'ineal.
Definimos:\,\\
\begin{equation*}
    \langle x,y\rangle'=\langle T(x),T(y)\rangle\,\,\,(x,y\in V) 
\end{equation*}
Demuestre que $\langle.,. \rangle'$ es producto interior en $V$ si y s\'olo si $T_F$ es inyectiva}
\end{tcolorbox}
\begin{proof}\,\\
	\,\\
    Supongamos que $\langle,\rangle'$ es un producto interior en $V$. Sean $\{ x,y\}\subset _FV$, tales que $T(x)=T(y)$, entonces:
\begin{align*}
    &T(x-y)=T(x)-T(y)=0\\
    &\Rightarrow \langle T(x-y),T(x-y) \rangle=0\\
    &\Rightarrow \langle x-y,x-y \rangle \text{'}=0 \\
    &\Rightarrow x=y
\end{align*}
Es decir $T$ es inyectiva.\\[10pt]
Ahora supongamos que $T$ es inyectiva, entonces:
\begin{align*}
   \forall \{x,y\}\subset V \quad \langle x,y \rangle'=\langle T(x),T(y) \rangle =\overline{\langle T(y),T(x) \rangle}=\overline{\langle y,x \rangle'}\\
\end{align*}
Dados $x,y,z\in V$ y $c\in F$, se tiene que:
\begin{align*}
    \langle x+cy,z \rangle'&=\langle T(x+cy),T(z) \rangle \\
    &=\langle T(x)+cT(y),T(z) \rangle \\
    &=\langle T(x),T(z) \rangle +c\langle T(y),T(z) \rangle \\
    &=\langle x,z \rangle' + c\langle y,z \rangle'
\end{align*}
Finalmente
\begin{align*}
     &\forall x\in V \quad \langle x,x \rangle'=\langle T(x),T(x) \rangle \in \R^{\leq 0}\\
     &0=\langle x,x \rangle'=\langle T(x),T(x) \rangle \Leftrightarrow T(x)=0 \Leftrightarrow x=0
\end{align*}
Pues $T$ es inyectiva. Por lo que $\langle,\rangle'$ es un producto interior en $V$.
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 3}},]
\textit{Pruebe que para cualesquiera $x,y\in V$:\,\\
\begin{align*}
    &i)\|x\pm y\|^2=\|x\|^2\pm Re\langle x,y\rangle+\|y\|^2\,\\
    \,\\
    &ii)|\,\|x\|-\|y\|\,|\leq \|x+y\|
\end{align*}
}
\end{tcolorbox}
\begin{proof}\,\\
	\,\\
	i)\,Tenemos que:\,\\
	\begin{align*}
		\|x+y\|^2&=\langle x+y,x+y\rangle=\langle x,x\rangle+\langle x,y\rangle+\langle y,x\rangle+\langle y,y\rangle\,\\
		\,\\
		&=\|x\|^2+\langle x,y\rangle+\overline{\langle x,y\rangle}+\|y\|^2\,\\
		\,\\
		&=\|x\|^2+Re\,\langle x,y\rangle+\|y\|^2
	\end{align*}\,\\
	De la misma manera:\,\\
	\begin{align*}
		\|x-y\|^2&=\langle x-y,x-y\rangle=\langle x,x\rangle+\langle x,-y\rangle+\langle -y,x\rangle+\langle y,y\rangle\,\\
		\,\\
		&=\|x\|^2-\langle x,y\rangle+(\overline{-1})\overline{\langle x,y\rangle}+\|y\|^2\,\\
		\,\\
		&=\|x\|^2-\langle x,y\rangle-\overline{\langle x,y\rangle}+\|y\|^2\,\\
		\,\\
		&=\|x\|^2-Re\,\langle x,y\rangle+\|y\|^2
	\end{align*}\,\\
	\,\\
	ii)\,Aplicando la desigualdad del tri\'angulo:\,\\
	\begin{equation*}
		\forall x,y\in V\,(\|x\|=\|x+y-y\|\leq\|x+y\|+\|-y\|=\|x+y\|+\|y\|\implies\|x\|-\|y\|\leq \|x+y\|)
	\end{equation*}\,\\
	De esto se sigue que:\,\\
	\begin{equation*}
		\forall x,y\in V(\|y\|\leq\|x\|+\|y+x\|\implies-\|x+y\|\leq\|x\|-\|y\|)
	\end{equation*}\,\\
	\,\\
	Combinando ambas obtenemos:\,\\
	\begin{equation*}
		\forall x,y\in V (|\|x\|-\|y\||\leq\|x+y\|)
	\end{equation*}
\end{proof}\,\\

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 4}},]
\textit{Si $\{x_i\}_{i\in\N_k}\subset V$ es ortogonal y $\{a_i\}_{i\in \N_k}\subset F$, demuestre que:\,\\
\,\\
\begin{equation*}
	\left\|\sum_{i\in\N_k}\,a_i\,x_i\right\|^2=\sum_{i\in \N_k}|a_i|^2\|x_i\|^2
\end{equation*} 
}
\end{tcolorbox}
\begin{proof}\,\\
	\,\\
	Tenemos que:\,\\
	\,\\
	\begin{equation*}
		\left\|\sum_{i\in \N_k}\,a_i\,x_i \right\|^2=\left\langle \sum_{i\in \N_k}\,a_i\,x_i,\sum_{j\in \N_k}\,a_j\,x_j \right\rangle
	\end{equation*}\,\\
	\,\\
	Ahora definimos la siguiente funci\'on $\delta_{i,j}:\N_k\times \N_k\rightarrow \R$:\,\\
	\,\\
	\[
	\delta_{i,j} = 
	\begin{cases}
	0 & \text{si } i\neq j  \\
	\langle x_i,x_i \rangle& \text{si } i=j\\
	\end{cases}
	\]
	\,\\
	Aplicamos la aditividad de ambas entradas del producto interior:\,\\
	\,\\
	\begin{equation*}
		\left\|\sum_{i\in \N_k},a_i\,x_i \right\|^2=\sum_{i\in \N_k}\sum_{j\in \N_k}\langle a_i\,x_i,a_j\,x_j \rangle
	\end{equation*}\,\\
	\,\\
	Ahora aplicando la linealidad en la primera entrada y sacando el conjugado de la segunda, ademas usando que $\{x_i\}_{i\in \N_k}$ es ortogonal:\,\\
	\,\\
	\begin{equation*}
		\left\|\sum_{i\in \N_k},a_i\,x_i \right\|^2=\sum_{i\in\N_k}\,a_i\,\overline{a_i}\langle x_i,x_i\rangle=\sum_{i\in \N_k}\sum_{j\in \N_k}a_i\,\overline{a_j}\,\delta_{i,j}=\sum_{i\in \N_k}\,a_i\,\overline{a_i}\,\|x_i\|^2=\sum_{i\in \N_k}|a_i|^2\,\|x_i\|^2
	\end{equation*}\,\\
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 5 (convexidad de productos interiores)}},]
\textit{Sean $\langle.,. \rangle_1$ $\langle.,. \rangle_2$ productos interiores en $V$, sean $\alpha,\beta>0$.Muestre que:\,\\
\,\\
\begin{equation*}
	\langle x,y\rangle:=\alpha\,\langle x,y \rangle_1+\beta\langle x,y \rangle_2
\end{equation*}\,\\
define otro producto en $V$
}
\end{tcolorbox}
\begin{proof}\,\\
	\,\\
	i)\,Tenemos que $\langle.,y\rangle$ es lineal para todo $y\in V$, pues combianci\'on lineal de funciones lineales es lineal\,\\
	\,\\
	ii)\,Tenemos que:\,\\
	\,\\
	\begin{equation*}
		\overline{\langle x,y\rangle}=\overline{\alpha\,\langle x,y\rangle_1+\beta\,\langle x,y\rangle_2}=\overline{\alpha}\,\overline{\langle x,y \rangle_1}+\overline{\beta}\,\,\overline{\langle x,y\rangle_2}=\overline{\alpha}\,\langle y,x \rangle_1+\overline{\beta}\,\langle y,x\rangle_2
	\end{equation*}\,\\
	Como $\alpha,\beta\in \R^+$ entonces $\overline{\beta}=\beta$ y $\overline{\alpha}=\alpha$ se tiene que:\,\\
	\,\\
	\begin{equation*}
		\overline{\langle x,y\rangle}=\alpha\,\langle y,x \rangle_1+\beta\,\langle y,x\rangle_2=\langle y,x\rangle
	\end{equation*}\,\\
	iii)\,Tenemos que:\,\\
	\begin{equation*}
	\forall x\in V	(\langle x,x\rangle=\alpha\,\langle x,x\rangle_1+\beta\langle x,x\rangle_2\geq 0)
	\end{equation*}\,\\
	iv)\,Finalmente:\,\\
	\begin{equation*}
		\langle x,x\rangle=0\iff (\langle x,x\rangle_1=0\,\,\,y\,\,\,\langle x,x\rangle_2=0)\iff x=0_V
	\end{equation*}\,\\
	Pues por iii) $\langle x,x \rangle\geq 0$, por tanto la unica manera en que el producto interior definido sea $0$ es que sus dos componentes mayores a cero sean $0$

\end{proof}\,\\

}
\end{tcolorbox}

\end{proof}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 7}},]
\textit{Pruebe que en cualquier espacio con producto interior de cumple que:\,\\
\,\\
\begin{equation*}
	\|x+y\|^2+\|x-y\|^2=2\|x\|^2+2\|y\|^2
\end{equation*}
}
\end{tcolorbox}
\begin{proof}\,\\
	\,\\
Sean $\{x,y\}\in _FV$. Se tiene que:
\begin{align*}
    \| x + y \|^2&=\langle x+y,x+y \rangle\\
    &=\langle x,x \rangle + \langle x,y \rangle + \langle y,x \rangle + \langle y,y \rangle\\
    &=\|x\|^2 + \langle x,y \rangle + \langle y,x \rangle + \|y\|^2\\
    \| x - y \|^2&=\langle x-y,x-y \rangle\\
    &=\langle x,x \rangle + \langle x,-y \rangle + \langle -y,x \rangle + \langle -y,-y \rangle\\
    &=\|x\|^2 + \bar{(-1)} \langle x,y \rangle + (-1)\langle y,x \rangle + \|y\|^2\\
\end{align*}
Por lo tanto:
\begin{align*}
    \| x + y \|^2 + \| x - y \|^2 = 2\|x\|^2 + 2\|y\|^2 + \langle x,y \rangle + \langle y,x \rangle - \langle x,y \rangle - \langle y,x \rangle= 2\|x\|^2 + 2\|y\|^2
\end{align*}
\end{proof}

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 9}},]
\textit{Sea $W$ un subespacio de dimensi\'on finita de $V$. Demuestre que para todo $x\in V$ existe un \'unico $y\in W$ tal que $x-y\in W^{\bot}$. Ademas demuestre 
que el operador $P:V\rightarrow W$, $P(x)=y$, es lineal y satisface $P^2=P$ y $\|P\|\leq 1$. Donde:\,\\
\,\\
\begin{equation*}
	\|P\|=\sup_{x\neq 0_v}\,\frac{\|P(x)\|}{\|x\|} 
\end{equation*} 
}
\end{tcolorbox}
\begin{proof}\,\\
	\,\\
	Como $W$ es subespacio de dimensi\'on finita de $V$, existe su complemento ortogonal $W^{\perp}$, como\\ $V=W\oplus W^{\perp}$, entonces:\,\\
	\,\\
	\begin{equation*}
		\forall x\in V\,\,\,\exists\,\,!w^{\perp}\in W^{\perp},\,\,w\in W\,\,\,(x=w+w^{\perp})
	\end{equation*}\,\\
	de aqui tenemos que:\,\\
	\,\\
	\begin{equation*}
		\forall x\in V\,\,\exists\,\,!w\in W\,\,\,(x-w=w^{\perp}+w-w=w^{\perp}\in W^{\perp})
	\end{equation*}\,\\
	Ahora si tomamos $w\in W$, entonces $w=0_{v}+w$, debido a que esta descomposici\'on es unica pues $0_V\in W^{\perp}$, se sigue que:\,\\
	\,\\
	\begin{equation*}
		\forall\,w\in W\,\,(P(w)=w)
	\end{equation*}\,\\
	Por tanto si tomamos $v\in V$, se sigue que $P(v)\in W$, luego:\,\\
	\,\\
	\begin{equation*}
		\forall v\in V\,\,(P(P(v))=P(v))
	\end{equation*}\,\\
	Luego tomamos $v_1,v_2\in V$, sabemos que existen unicos $w_1,w_2\in W$ y $w_1^{\perp},w_2^{\perp}\in W^{\perp}$ tales que $v_1=w_1+w_1^{\perp}$ y 
	$v_2=w_2+w_2^{\perp}$, luego se tiene que $v_1+v_2=(w_1+w_2)+(w_1^{\perp}+w_2^{\perp})$, luego:\,\\
	\,\\
	\begin{equation*}
		P(v_1+v_2)=w_1+w_2
	\end{equation*}\,\\
	Pues $(w_1+w_2)\in W$ y $(w_1^{\perp}+w_2^{\perp})\in W^{\perp}$, finalemente:\,\\
	\,\\
	\begin{equation*}
		P(v_1+v_2)=w_1+w_2=P(v_1)+P(v_2)
	\end{equation*}\,\\
	An\'alogamente sea $\alpha \in \mathbb{F}$ entonces $\alpha\,v_1=\alpha\,w_1+\alpha\,w_2$, entonces:\,\\
	\,\\
	\begin{equation*}
		P(\alpha\,v_1)=\alpha\,w_1
	\end{equation*}\,\\
	Pues $(\alpha\,w_1)\in W$ y $(\alpha\,w^{\perp})\in W^{\perp}$, se obtiene:\,\\
	\,\\
	\begin{equation*}
		P(\alpha\,v_1)=\alpha\,w_1=\alpha\,P(v_1)
	\end{equation*}\,\\
	Luego si tomamos $x\neq 0_{V}$ se sigue que $\|x\|\neq 0$, luego sabemos que en particular existen $w\in W$ y $w^{\perp}\in W^{\perp}$ unicos tales
	que $x=w+w^{\perp}$, donde $w\neq 0_V$ \'o $w^{\perp}\neq 0_V$:\,\\
	\,\\
	\begin{align*}
		\|x\|^2&=\langle w+w^{\perp},w+w^\perp\rangle=\langle w,w \rangle+\langle w^{\perp},w \rangle+\langle w,w^{\perp} \rangle+\langle w^{\perp},w^{\perp}\rangle=\langle w^{\perp},w^{\perp}\rangle+\langle w,w\rangle\\
		\,\\
		&=\|w\|^2+\|w^{\perp}\|^2
	\end{align*}\,\\
	Por tanto:\,\\
	\begin{equation*}
		\frac{\|P(x)\|}{\|x\|}=\frac{\|w\|}{\sqrt{\|w\|^2+\|w^{\perp}\|^2}}\leq 1
	\end{equation*}\,\\
	Finalmente concluimos que:\,\\
	\,\\
	\begin{equation*}
		\sup_{x\neq 0_V}\,\frac{\|P(x)\|}{\|x\|}\leq 1
	\end{equation*}
\end{proof}\,\\

\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 10}},]
\textit{Dado el conjunto linealmente independientes $S:=\{(1,0,0),(1,2,4),(1,2,1)\}\subset \R^3$:\,\\
\,\\
i)\,Aplique el m\'etodo de Gram-Schmidt para obtener una base ortonormal $\{u_1,u_2,u_3\}$ de $L(S)$\,\\
\,\\
ii)Exprese a cada vector de S como combinaci\'on l\'ineal de $\{u_1,u_2,u_3\}$
}
\end{tcolorbox}\,\\

Sean $w_1=\begin{bmatrix} 1\\0\\0 \end{bmatrix}, w_2=\begin{bmatrix} 1\\2\\4 \end{bmatrix}, w_3=\begin{bmatrix} 1\\2\\1 \end{bmatrix}$. Entonces:
\begin{align*}
    u_1&=w_1=\begin{bmatrix} 1\\0\\0 \end{bmatrix}\\
\end{align*}
Normalizamos $u_1$:
\begin{align*}
    v_1&=\frac{u_1}{\|u_1\|}=\frac{1}{1}\begin{bmatrix} 1\\0\\0 \end{bmatrix}
\end{align*}
Calculamos $u_2$:
\begin{align*}
    u_2&=w_2-\langle w_2,v_1 \rangle v_1\\
    &=\begin{bmatrix} 1\\2\\4 \end{bmatrix}-\left\langle \begin{bmatrix} 1\\2\\4 \end{bmatrix},\begin{bmatrix} 1\\0\\0 \end{bmatrix} \right\rangle \begin{bmatrix} 1\\0\\0 \end{bmatrix}\\
    &=\begin{bmatrix} 1\\2\\4 \end{bmatrix}-\begin{bmatrix} 1\\0\\0 \end{bmatrix}\\
    &=\begin{bmatrix} 0\\2\\4 \end{bmatrix}
\end{align*}
Normalizamos $u_2$:
\begin{align*}
v_2&=\frac{u_1}{\|u_2\|}=\frac{1}{\sqrt{20}}\begin{bmatrix} 0\\2\\4 \end{bmatrix}=\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix}
\end{align*}
Calculamos $u_3$:
\begin{align*}
    u_3&=w_3-\langle w_3,v_2 \rangle v_2 -\langle w_3,v_1 \rangle v_1\\
    &=\begin{bmatrix} 1\\2\\1 \end{bmatrix}-\left\langle \begin{bmatrix} 1\\2\\1 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix} \right\rangle \frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix}
    -\left\langle \begin{bmatrix} 1\\2\\1 \end{bmatrix},\begin{bmatrix} 1\\0\\0 \end{bmatrix} \right\rangle \begin{bmatrix} 1\\0\\0 \end{bmatrix}\\
    &=\begin{bmatrix} 1\\2\\4 \end{bmatrix}-\frac{4}{5}\begin{bmatrix} 0\\1\\2 \end{bmatrix}-\begin{bmatrix} 1\\0\\0 \end{bmatrix}\\
    &=\frac{3}{5}\begin{bmatrix} 0\\2\\-1 \end{bmatrix}
\end{align*}
Normalizamos $u_3$:
\begin{align*}
v_3&=\frac{u_1}{\|u_2\|}=\left(\frac{\sqrt{5}}{3}\right)\frac{3}{5}\begin{bmatrix} 0\\2\\-1 \end{bmatrix}=\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\2\\-1 \end{bmatrix}
\end{align*}
Comprobamos que $\{v_1,v_2,v_3\}$ es ortogonal:
\begin{align*}
    &\langle v_1,v_2 \rangle = \left\langle \begin{bmatrix} 1\\0\\0 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix} \right\rangle = 0\\
    &\langle v_1,v_3 \rangle = \left\langle \begin{bmatrix} 1\\0\\0 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\2\\-1 \end{bmatrix} \right\rangle = 0\\
    &\langle v_2,v_3 \rangle = \left\langle \frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\2\\-1 \end{bmatrix} \right\rangle = 0
\end{align*}
Luego $\{v_1,v_2,v_3\}$ es ortonormal.\\[10pt]
Ahora calculemos:
\begin{align*}
    \langle w_1,v_1\rangle=\left\langle \begin{bmatrix} 1\\0\\0 \end{bmatrix},\begin{bmatrix} 1\\0\\0 \end{bmatrix} \right\rangle=1, 
    &&\langle w_1,v_2\rangle=\left\langle \begin{bmatrix} 1\\0\\0 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix} \right\rangle =0,
    &&\langle w_1,v_3\rangle=\left\langle \begin{bmatrix} 1\\0\\0 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\2\\-1 \end{bmatrix} \right\rangle = 0. \\
    \langle w_2,v_1\rangle=\left\langle \begin{bmatrix} 1\\2\\4 \end{bmatrix},\begin{bmatrix} 1\\0\\0 \end{bmatrix} \right\rangle=1, 
    &&\langle w_2,v_2\rangle=\left\langle \begin{bmatrix} 1\\2\\4 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix} \right\rangle =\frac{10}{\sqrt{5}},
    &&\langle w_2,v_3\rangle=\left\langle \begin{bmatrix} 1\\2\\4 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\2\\-1 \end{bmatrix} \right\rangle =0.\\
     \langle w_3,v_1\rangle=\left\langle \begin{bmatrix} 1\\2\\1 \end{bmatrix},\begin{bmatrix} 1\\0\\0 \end{bmatrix} \right\rangle=1, 
     &&\langle w_3,v_2\rangle=\left\langle \begin{bmatrix} 1\\2\\1 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix} \right\rangle=\frac{4}{\sqrt{5}}, 
     &&\langle w_3,v_3\rangle=\left\langle \begin{bmatrix} 1\\2\\1 \end{bmatrix},\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\2\\-1 \end{bmatrix} \right\rangle=\frac{3}{\sqrt{5}}.
\end{align*}

\begin{align*}
    1v_1 + 0v_2 + 0v_3 &= \begin{bmatrix} 1\\0\\0 \end{bmatrix} = w_1.\\
    1v_1 + \frac{10}{\sqrt{5}}v_2 + 0v_3 &=\begin{bmatrix} 1\\0\\0 \end{bmatrix} + \left(\frac{10}{\sqrt{5}}\right)\frac{1}{\sqrt{5}}\begin{bmatrix} 0\\1\\2 \end{bmatrix}= \begin{bmatrix} 1\\2\\4 \end{bmatrix} = w_2.\\

\end{align*}
