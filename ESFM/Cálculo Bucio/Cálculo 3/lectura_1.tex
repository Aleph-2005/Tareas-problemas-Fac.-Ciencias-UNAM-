\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 1}}]
\textit{\textbf{Demuestre} que la hip\'otesis:\,\\
\begin{equation*}
    \forall x\in \R^n\,\,(\parallel x \parallel \geq 0)
\end{equation*}\,\\
en la definici\'on de norma es redundante.
}
\end{tcolorbox}
\begin{proof}\,\\
    \,\\
    Tomemos $x\in \R^n$ se sigue que:\,\\
    \begin{equation*}
        0=\parallel x-x \parallel \leq\parallel x \parallel+\parallel-x\parallel=2\parallel x \parallel \implies 0\leq \parallel x \parallel
    \end{equation*}
\end{proof}
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 2}}]
\textit{Sea $T:\R^n\rightarrow \R^n$ una transformaci\'on l\'ineal invertible, sea $\{v_1,v_2,\cdots,v_n\}$ una base de
$\R^n$ y sea $\parallel.\parallel$ una norma sobre $\R^n$.\textbf{Pruebe} que la funci\'on:\,\\
\begin{equation*}
    \forall\,x=\sum_{i=1}^n\,\alpha_i\,v_i \in \R^n\,\,\left(\parallel\,x\,\parallel' =\sum_{i=1}^n\,\parallel \alpha_i T(v_i) \parallel\right)
\end{equation*}
Es una norma en $\R^n$
}
\end{tcolorbox}\,\\
\begin{proof}\,\\
    \,\\
    \textbf{I)}\,\,\,$\forall x \in \R^n$\,\,\,$(\parallel x \parallel'\geq 0)$ y $\parallel x \parallel'=0\iff x=0_{\R^n}$
    \,\\
    \begin{equation*}
        \parallel x \parallel=\sum_{i=1}^n \parallel \alpha_i T(v_i) \parallel =\sum_{i=1}^n |\alpha_i|\,\parallel T(v_i) \parallel
    \end{equation*}\,\\
    \,\\
    Como $\parallel . \parallel$ es una norma de $\R^n$ entonces $\parallel T(v_i) \parallel\geq 0$ $\forall i\in \N_n$, por tanto:\,\\
    \begin{equation*}
        \forall\,x\in \R^n\,\,(\parallel x \parallel\geq 0)
    \end{equation*}\,\\
    Luego si $x=0$, entonces $0=\alpha_i\,T(v_i)$, se cumple si y solo si $\alpha_i=0$ $\forall i\in \N_i$, ya que como $T$ es un isomorfismo manda bases en bases, es decir $\{T(v_i)\}_{i\in \N_n}$
    es linealmente independiente, por tanto $\parallel 0 \parallel'=0$, por otro lado si $\parallel x \parallel'=0$, obtenemosque:\,\\
    \,\\
    \begin{equation*}
        \parallel 0 \parallel'=\sum_{i=1}^n\,|\alpha_i|\,\parallel T(v_i) \parallel
    \end{equation*}\,\\
    \,\\
    Como $T(v_i)\neq 0_{\R^n}$ $\forall i\in \N_n$ (debido a que el conjunto es l.i) se tiene que necesariamente $\alpha_i=0$, $\forall i\in \N_n$ es decir $x=0_{\R^n}$\,\\
    \,\\ 
    \textbf{II)}\,$\forall x,y \in \R^n$\,\,$(\parallel x+y \parallel'\leq \parallel x \parallel' +\parallel y \parallel')$\,\\
    \,\\
    Sea $x,y\in \R^n$, entonces $x+y=\displaystyle\sum_{i=1}^{n}\,\,(\alpha_i+\beta_i)\,T(v_i)$, entonces:\,\\
    \,\\
    \begin{equation*}
        \parallel x+y \parallel'=\sum_{i=1}^{n}\,|\alpha_i+\beta_i|\,\parallel T(v_i) \parallel \leq \sum_{i=1}^{n}\,(|\alpha_i|+|\beta_i|)\,\parallel T(v_i) \parallel= \parallel x \parallel' +\parallel y \parallel'
    \end{equation*}\,\\
    \textbf{iii)}\,\,$\forall x\in \R^n\,\,\alpha \in \R$  $(\parallel \alpha x \parallel'=|\alpha|\parallel x \parallel')$:\,\\
    \,\\
    Se sigue inmediatamente:\,\\
    \,\\
    \begin{equation*}
        \parallel \alpha x \parallel=\sum_{i=1}^{n}\,\alpha \parallel \lambda_i T(v_i) \parallel =\alpha \sum_{i=1}^{n}\,\parallel \lambda_i T(v_i) \parallel=|\alpha|\parallel x \parallel'
    \end{equation*}
\end{proof}\,\\
\begin{tcolorbox}[
	title = \textcolor{black}{\textcolor{white}{Problema 3}}]
\textit{\textbf{Demuestra} que :\,\\
\,\\
\begin{equation*}
    \forall x\in \R^n \left(\parallel x \parallel_2=\sqrt{\parallel x \parallel_1 \parallel x \parallel_{\infty}}\leq \frac{1+\sqrt{n}}{2}\,\parallel x \parallel_2\right)
\end{equation*}
}
\end{tcolorbox}\,\\
\begin{proof}
    Sea $x=(x_1,x_2,\cdots,x_n)\in \R^n$, sabemos que $|x_i|\leq \parallel x \parallel_{\infty}\implies x_i^2\leq |x_i| \parallel x \parallel_{\infty}$ $\forall i\in \N_n$, por tanto se sigue que:\,\\
    \,\\
    \begin{equation*}
        \sum_{i=1}^n\,x_i^2=\sum_{i=1}^n\,|x_i| \parallel x \parallel_{\infty}\implies \parallel x \parallel_2\leq \sqrt{\parallel x \parallel_1 \parallel x \parallel_{\infty}}
    \end{equation*}\,\\
    Luego por la desigualdad de cauchy-swartz sobre $1=(1,1,1,\cdots,1)\in \R^n$ y $|x|=(|x_1|,|x_2|,\cdots,|x_n|)\in \R^n$, tenemos que:\,\\
    \begin{equation*}
        <x,1>=\parallel x \parallel_1\leq \sqrt{n}\,\parallel x \parallel_2
    \end{equation*}\,\\
    Ademas que $\parallel x \parallel_{\infty}\leq \parallel x \parallel_2$, aplicando estas desigualdades y la desigualdad entre M.A y M.G, se sigue que:\,\\
    \,\\
    \begin{equation*}
        \sqrt{\parallel x \parallel_1 \parallel x \parallel_{\infty}}\leq \frac{\parallel x \parallel_{\infty} +\parallel x \parallel_{1}}{2}\leq \frac{1+\sqrt{n}}{2}\,\parallel x \parallel_2
    \end{equation*}
\end{proof}
